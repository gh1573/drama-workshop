   147→   147→async function recomputeProgress(runId: string): Promise<void> {
   148→   148→  const row = await loadRun(runId);
   149→   149→  const stages = parseStages(row.stages);
   150→   150→  const done = stages
   151→   151→    .filter((s) => s.status === "succeeded")
   152→   152→    .reduce(
   153→   153→      (sum, s) =>
   154→   154→        sum + (RUN_STAGES.find((d) => d.key === s.key)?.weight ?? 0),
   155→   155→      0
   156→   156→    );
   157→   157→  const total = RUN_STAGES.reduce((sum, d) => sum + d.weight, 0);
   158→   158→  await db.productionRun.update({
   159→   159→    where: { id: runId },
   160→   160→    data: { progress: Math.min(100, Math.round((done / total) * 100)) },
   161→   161→  });
   162→   162→}

   255→   255→function parseBeats(raw: string): StoryBeat[] | null {
   256→   256→  const data = parseJsonLoose(raw) as { storyBeats?: unknown[] } | null;
   257→   257→  const arr = data?.storyBeats;
   258→   258→  if (!Array.isArray(arr) || arr.length < 3) return null;
   259→   259→
   260→   260→  const funcs: NarrativeFunction[] = ["hook", "climax", "cliffhanger"];
   261→   261→  const beats: StoryBeat[] = [];
   262→   262→  for (let i = 0; i < 3; i++) {
   263→   263→    const item = arr[i] as Record<string, unknown>;
   264→   264→    const fn = funcs.includes(item?.narrativeFunction as NarrativeFunction)
   265→   265→      ? (item.narrativeFunction as NarrativeFunction)
   266→   266→      : funcs[i];
   267→   267→    const meta = NARRATIVE_META[fn];
   268→   268→    const str = (v: unknown, fallback: string) =>
   269→   269→      typeof v === "string" && v.trim() ? v.trim() : fallback;
   270→   270→    const cam =
   271→   271→      typeof item?.cameraMovement === "string" &&
   272→   272→      item.cameraMovement in CAMERA_MOVEMENTS
   273→   273→        ? item.cameraMovement
   274→   274→        : "";
   275→   275→    beats.push({
   276→   276→      index: i + 1,
   277→   277→      actTitle: `${meta.actNo}：${meta.label} (${meta.tag})`,
   278→   278→      narrativeFunction: fn,
   279→   279→      pacingNote: str(item?.pacingNote, PACING_FALLBACK[fn]),
   280→   280→      description: str(item?.description, "（分镜内容生成异常，请重试）"),
   281→   281→      dialogueSubtitle: str(item?.dialogueSubtitle, "台词生成异常，请重试"),
   282→   282→      cameraMovement: cam,
   283→   283→      promptGuidance: str(item?.promptGuidance, "提示生成异常，请重试"),
   284→   284→      energyStart: null,
   285→   285→      energyEnd: null,
   286→   286→    });
   287→   287→  }
   288→   288→  // 运镜合规修复：池外 → 本幕池；三幕互斥 → 换池内备选
   289→   289→  const used = new Set<string>();
   290→   290→  for (const beat of beats) {
   291→   291→    const pool = ACT_MOVEMENT_POOLS[beat.narrativeFunction];
   292→   292→    let mv = beat.cameraMovement;
   293→   293→    if (!pool.includes(mv)) mv = pool[0];
   294→   294→    if (used.has(mv)) {
   295→   295→      const alt = pool.find((k) => !used.has(k));
   296→   296→      if (alt) mv = alt;
   297→   297→    }
   298→   298→    used.add(mv);
   299→   299→    beat.cameraMovement = mv;
   300→   300→  }
   301→   301→  return beats;
   302→   302→}

   366→   366→【要求（严格遵守）】

   368→   368→2. scenes：关键场景，最多 5 个。name 为短名（如「宴会厅」「冷宫」）；lockedAppearance 描述空间与光线；visualAnchor 为英文
   369→   369→3. props：推动剧情的关键道具，最多 4 个，没有则输出空数组
   370→   370→4. 名称必须与剧本用词一致，不得杜撰剧本中未出现的角色
   371→   371→5. visualAnchor 是所有镜头共用的视觉锚点，聚焦「跨镜头不变的外观特征」，不要写动作
   372→   372→
   373→   373→【输出格式】只输出 JSON，禁止解释文字、禁止 markdown 围栏：
   374→   374→{"characters":[{"name":"…","lockedAppearance":"…","visualAnchor":"…"}],"scenes":[{"name":"…","lockedAppearance":"…","visualAnchor":"…"}],"props":[{"name":"…","lockedAppearance":"…","visualAnchor":"…"}]}`;
   375→   375→}
   376→   376→
   377→   377→function parseCasting(
   378→   378→  raw: string
   379→   379→): { characters: CastingAssetDraft[]; scenes: CastingAssetDraft[]; props: CastingAssetDraft[] } | null {
   380→   380→  const data = parseJsonLoose(raw) as Record<string, unknown> | null;
   381→   381→  if (!data) return null;
   382→   382→
   383→   383→  const map = (v: unknown, max: number): CastingAssetDraft[] =>
   384→   384→    (Array.isArray(v) ? v : [])
   385→   385→      .map((item) => {
   386→   386→        const o = item as Record<string, unknown>;
   387→   387→        const name = typeof o?.name === "string" ? o.name.trim().slice(0, 20) : "";
   388→   388→        if (!name) return null;
   389→   389→        const lockedAppearance =
   390→   390→          typeof o?.lockedAppearance === "string" ? o.lockedAppearance.trim().slice(0, 200) : "";
   391→   391→        const visualAnchor =
   392→   392→          typeof o?.visualAnchor === "string" ? o.visualAnchor.trim().slice(0, 300) : "";
   393→   393→        if (!lockedAppearance && !visualAnchor) return null;
   394→   394→        return {
   395→   395→          name,
   396→   396→          lockedAppearance: lockedAppearance || "（定妆描述待补充）",
   397→   397→          visualAnchor: visualAnchor || lockedAppearance,
   398→   398→        } satisfies CastingAssetDraft;
   399→   399→      })
   400→   400→      .filter((x): x is CastingAssetDraft => !!x)
   401→   401→      .slice(0, max);
   402→   402→
   403→   403→  const characters = map(data.characters, 6);
   404→   404→  const scenes = map(data.scenes, 5);
   405→   405→  const props = map(data.props, 4);
   406→   406→  if (characters.length === 0 && scenes.length === 0) return null;
   407→   407→  return { characters, scenes, props };
   408→   408→}

   573→   573→/** 按名称把拆镜产物中的资产引用解析为 ShotRef（精确名 → 包含式回退） */
   574→   574→export async function bindRefsFromShots(
   575→   575→  projectId: string,
   576→   576→  beatIndex: number,
   577→   577→  shots: ShotBreakdownItem[],
   578→   578→  clips: Array<{ id: string; beatIndex: number; order: number }>
   579→   579→): Promise<number> {
   580→   580→  const assets = await db.asset.findMany({ where: { projectId } });
   581→   581→  const byKind = (kind: AssetKind) => assets.filter((a) => a.kind === kind);
   582→   582→  const resolve = (kind: AssetKind, name: string) => {
   583→   583→    if (!name) return null;
   584→   584→    const pool = byKind(kind);
   585→   585→    return (
   586→   586→      pool.find((a) => a.name === name) ??
   587→   587→      pool.find(
   588→   588→        (a) => a.name.includes(name) || name.includes(a.name)
   589→   589→      ) ??
   590→   590→      null
   591→   591→    );
   592→   592→  };
   593→   593→
   594→   594→  const actClips = clips
   595→   595→    .filter((c) => c.beatIndex === beatIndex)
   596→   596→    .sort((a, b) => a.order - b.order);
   597→   597→  let bound = 0;
   598→   598→  for (let i = 0; i < actClips.length && i < shots.length; i++) {
   599→   599→    const shot = shots[i];
   600→   600→    const clip = actClips[i];
   601→   601→    const targets = new Set<string>();
   602→   602→    for (const name of shot.characters) {
   603→   603→      const asset = resolve("character", name);
   604→   604→      if (asset) targets.add(asset.id);
   605→   605→    }
   606→   606→    const scene = resolve("scene", shot.scene);
   607→   607→    if (scene) targets.add(scene.id);
   608→   608→    for (const name of shot.props) {
   609→   609→      const asset = resolve("prop", name);
   610→   610→      if (asset) targets.add(asset.id);
   611→   611→    }
   612→   612→    for (const assetId of targets) {
   613→   613→      await db.shotRef.upsert({
   614→   614→        where: { shotId_assetId: { shotId: clip.id, assetId } },
   615→   615→        create: { shotId: clip.id, assetId },
   616→   616→        update: {},
   617→   617→      });
   618→   618→      bound++;
   619→   619→    }
   620→   620→  }
   621→   621→  return bound;
   622→   622→}

   628→   628→export async function anchorProjectClips(
   629→   629→  projectId: string,
   630→   630→  onlyClipIds?: string[]
   631→   631→): Promise<{ anchored: number; skipped: number }> {
   632→   632→  const [project, clips, refs] = await Promise.all([
   633→   633→    db.project.findUnique({ where: { id: projectId } }),
   634→   634→    db.timelineClip.findMany({
   635→   635→      where: { projectId, ...(onlyClipIds ? { id: { in: onlyClipIds } } : {}) },
   636→   636→      orderBy: { order: "asc" },
   637→   637→    }),
   638→   638→    db.shotRef.findMany({
   639→   639→      where: { shot: { projectId } },
   640→   640→      include: { asset: true },
   641→   641→    }),
   642→   642→  ]);
   643→   643→  if (!project) throw new RouteError("项目不存在或已被删除", 404);
   644→   644→
   645→   645→  const refsByClip = new Map<string, typeof refs>();
   646→   646→  for (const r of refs) {
   647→   647→    const list = refsByClip.get(r.shotId) ?? [];
   648→   648→    list.push(r);
   649→   649→    refsByClip.set(r.shotId, list);
   650→   650→  }
   651→   651→
   652→   652→  let anchored = 0;
   653→   653→  let skipped = 0;
   654→   654→  for (const clip of clips) {
   655→   655→    if (clip.promptVersion === "manual") {
   656→   656→      skipped++;
   657→   657→      continue;
   658→   658→    }
   659→   659→    const bound = refsByClip.get(clip.id) ?? [];
   660→   660→    const characters = bound
   661→   661→      .filter((r) => r.asset.kind === "character")
   662→   662→      .map((r) => ({
   663→   663→        name: r.asset.name,
   664→   664→        visualAnchor: r.asset.visualAnchor,
   665→   665→        stateNote: r.stateNote,
   666→   666→      }));
   667→   667→    const sceneRef = bound.find((r) => r.asset.kind === "scene");
   668→   668→    const props = bound
   669→   669→      .filter((r) => r.asset.kind === "prop")
   670→   670→      .map((r) => ({
   671→   671→        name: r.asset.name,
   672→   672→        visualAnchor: r.asset.visualAnchor,
   673→   673→        stateNote: r.stateNote,
   674→   674→      }));
   675→   675→
   676→   676→    const { finalPrompt, fingerprint } = composeVideoPrompt({
   677→   677→      shot: {
   678→   678→        visualDesc: clip.visualDesc ?? clip.name,
   679→   679→        visualPrompt: clip.visualPrompt,
   680→   680→        cameraMovement: clip.cameraMovement,
   681→   681→        shotSize: clip.shotSize,
   682→   682→        subtitle: clip.subtitle,
   683→   683→      },
   684→   684→      characters,
   685→   685→      scene: sceneRef
   686→   686→        ? {
   687→   687→            name: sceneRef.asset.name,
   688→   688→            visualAnchor: sceneRef.asset.visualAnchor,
   689→   689→            stateNote: sceneRef.stateNote,
   690→   690→          }
   691→   691→        : null,
   692→   692→      props,
   693→   693→      tonePrompt: project.tonePrompt,
   694→   694→      aspectRatio: project.aspectRatio,
   695→   695→      plannedMs: clip.plannedMs,
   696→   696→      resolution: project.resolution,
   697→   697→    });
   698→   698→
   699→   699→    await db.timelineClip.update({
   700→   700→      where: { id: clip.id },
   701→   701→      data: { finalPrompt, promptVersion: fingerprint },
   702→   702→    });
   703→   703→    anchored++;
   704→   704→  }
   705→   705→  return { anchored, skipped };
   706→   706→}

   755→   755→async function executeRun(runId: string): Promise<void> {
   756→   756→  try {
   757→   757→    const run = await db.productionRun.findUnique({ where: { id: runId } });
   758→   758→    if (!run || run.status !== "running") return;
   759→   759→
   760→   760→    for (const def of RUN_STAGES) {
   761→   761→      const current = await loadRun(runId);
   762→   762→      if (current.status !== "running") return; // 被外部取消/覆盖
   763→   763→      const stages = parseStages(current.stages);
   764→   764→      const stage = stages.find((s) => s.key === def.key);
   765→   765→      if (!stage || stage.status === "succeeded") continue;
   766→   766→
   767→   767→      // —— 阶段置 running ——
   768→   768→      stage.status = "running";
   769→   769→      stage.startedAt = new Date().toISOString();
   770→   770→      stage.error = null;
   771→   771→      await patchStages(runId, stages);
   772→   772→      await db.productionRun.update({
   773→   773→        where: { id: runId },
   774→   774→        data: { stage: def.key },
   775→   775→      });
   776→   776→      const project = await db.project.findUnique({
   777→   777→        where: { id: current.projectId },
   778→   778→      });
   779→   779→      if (!project) throw new Error("项目不存在或已被删除");
   780→   780→
   781→   781→      const startMsg =
   782→   782→        STAGE_START_MESSAGE[def.key] ?? `进入「${def.name}」阶段`;
   783→   783→      const modeLabel =
   784→   784→        def.key === "screenwriting" ? thinkingModeLabel(project.thinkingLevel) : "";
   785→   785→      await appendLog(
   786→   786→        runId,
   787→   787→        def.role,
   788→   788→        startMsg.replace("{mode}", modeLabel)
   789→   789→      );
   790→   790→
   791→   791→      try {
   792→   792→        const result = await STAGE_FN[def.key](runId, project);
   793→   793→        const fresh = parseStages((await loadRun(runId)).stages);
   794→   794→        const stageRow = fresh.find((s) => s.key === def.key);
   795→   795→        if (stageRow) {
   796→   796→          stageRow.status = "succeeded";
   797→   797→          stageRow.finishedAt = new Date().toISOString();
   798→   798→          await patchStages(runId, fresh);
   799→   799→        }
   800→   800→        await appendLog(runId, def.role, result.summary);
   801→   801→        await recomputeProgress(runId);
   802→   802→      } catch (err) {
   803→   803→        const message =
   804→   804→          err instanceof Error ? err.message : "未知错误";
   805→   805→        const fresh = parseStages((await loadRun(runId)).stages);
   806→   806→        const stageRow = fresh.find((s) => s.key === def.key);
   807→   807→        if (stageRow) {
   808→   808→          stageRow.status = "failed";
   809→   809→          stageRow.finishedAt = new Date().toISOString();
   810→   810→          stageRow.error = message;
   811→   811→          await patchStages(runId, fresh);
   812→   812→        }
   813→   813→        await db.productionRun.update({
   814→   814→          where: { id: runId },
   815→   815→          data: { status: "failed", error: `「${def.name}」阶段失败：${message}` },
   816→   816→        });
   817→   817→        await appendLog(runId, def.role, `阶段失败：${message}`);
   818→   818→        return;
   819→   819→      }
   820→   820→    }
   821→   821→
   822→   822→    await db.productionRun.update({
   823→   823→      where: { id: runId },
   824→   824→      data: { status: "succeeded", progress: 100 },
   825→   825→    });
   826→   826→    await appendLog(
   827→   827→      runId,
   828→   828→      "producer",
   829→   829→      "全片杀青 🎬 制作包已就绪：三幕剧本 · 分镜表 · 资产定妆 · 镜头级提示词 · 三轨 · 能量曲线，可导出交付"
   830→   830→    );
   831→   831→  } catch (err) {
   832→   832→    const message = err instanceof Error ? err.message : "未知错误";
   833→   833→    await db.productionRun
   834→   834→      .update({
   835→   835→        where: { id: runId },
   836→   836→        data: { status: "failed", error: message },
   837→   837→      })
   838→   838→      .catch(() => undefined);
   839→   839→  }
   840→   840→}

   879→   879→export async function retryRun(runId: string): Promise<RunDTO> {
   880→   880→  const run = await loadRun(runId);
   881→   881→  if (run.status === "running") {
   882→   882→    throw new RouteError("流水线正在运行中，无需重试", 409);
   883→   883→  }
   884→   884→  const stages = parseStages(run.stages).map((s) =>
   885→   885→    s.status === "failed"
   886→   886→      ? { ...s, status: "pending", error: null, startedAt: null, finishedAt: null }
   887→   887→      : s
   888→   888→  );
   889→   889→  const resume = stages.find((s) => s.status !== "succeeded");
   890→   890→  await db.productionRun.update({
   891→   891→    where: { id: runId },
   892→   892→    data: {
   893→   893→      status: "running",
   894→   894→      error: null,
   895→   895→      stages: JSON.stringify(stages),
   896→   896→    },
   897→   897→  });
   898→   898→  await appendLog(
   899→   899→    runId,
   900→   900→    "producer",
   901→   901→    resume
   902→   902→      ? `断点续拍：从「${resume.name}」阶段继续（已完成阶段不重复执行）`
   903→   903→      : "复核完成，直接进入收尾"
   904→   904→  );
   905→   905→  void executeRun(runId);
   906→   906→  const fresh = await loadRun(runId);
   907→   907→  return toRunDTO(fresh);
   908→   908→}
