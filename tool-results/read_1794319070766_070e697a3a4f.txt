    58→function parseStages(raw: string): RunStageInfo[] {
    59→  try {
    60→    const parsed = JSON.parse(raw) as RunStageInfo[];
    61→    if (Array.isArray(parsed) && parsed.length === RUN_STAGES.length) {
    62→      return parsed;
    63→    }
    64→  } catch {
    65→    /* fallthrough */
    66→  }
    67→  return freshStages();
    68→}

    71→  try {
    72→    const parsed = JSON.parse(raw) as RunLogEntry[];
    73→    if (Array.isArray(parsed)) return parsed;
    74→  } catch {
    75→    /* fallthrough */
    76→  }
    77→  return [];
    78→}

   146→async function recomputeProgress(runId: string): Promise<void> {
   147→  const row = await loadRun(runId);
   148→  const stages = parseStages(row.stages);
   149→  const done = stages
   150→    .filter((s) => s.status === "succeeded")
   151→    .reduce(
   152→      (sum, s) =>
   153→        sum + (RUN_STAGES.find((d) => d.key === s.key)?.weight ?? 0),
   154→      0
   155→    );
   156→  const total = RUN_STAGES.reduce((sum, d) => sum + d.weight, 0);
   157→  await db.productionRun.update({
   158→    where: { id: runId },
   159→    data: { progress: Math.min(100, Math.round((done / total) * 100)) },
   160→  });
   161→}

   254→function parseBeats(raw: string): StoryBeat[] | null {
   255→  const data = parseJsonLoose(raw) as { storyBeats?: unknown[] } | null;
   256→  const arr = data?.storyBeats;
   257→  if (!Array.isArray(arr) || arr.length < 3) return null;
   258→
   259→  const funcs: NarrativeFunction[] = ["hook", "climax", "cliffhanger"];
   260→  const beats: StoryBeat[] = [];
   261→  for (let i = 0; i < 3; i++) {
   262→    const item = arr[i] as Record<string, unknown>;
   263→    const fn = funcs.includes(item?.narrativeFunction as NarrativeFunction)
   264→      ? (item.narrativeFunction as NarrativeFunction)
   265→      : funcs[i];
   266→    const meta = NARRATIVE_META[fn];
   267→    const str = (v: unknown, fallback: string) =>
   268→      typeof v === "string" && v.trim() ? v.trim() : fallback;
   269→    const cam =
   270→      typeof item?.cameraMovement === "string" &&
   271→      item.cameraMovement in CAMERA_MOVEMENTS
   272→        ? item.cameraMovement
   273→        : "";
   274→    beats.push({
   275→      index: i + 1,
   276→      actTitle: `${meta.actNo}：${meta.label} (${meta.tag})`,
   277→      narrativeFunction: fn,
   278→      pacingNote: str(item?.pacingNote, PACING_FALLBACK[fn]),
   279→      description: str(item?.description, "（分镜内容生成异常，请重试）"),
   280→      dialogueSubtitle: str(item?.dialogueSubtitle, "台词生成异常，请重试"),
   281→      cameraMovement: cam,
   282→      promptGuidance: str(item?.promptGuidance, "提示生成异常，请重试"),
   283→      energyStart: null,
   284→      energyEnd: null,
   285→    });
   286→  }
   287→  // 运镜合规修复：池外 → 本幕池；三幕互斥 → 换池内备选
   288→  const used = new Set<string>();
   289→  for (const beat of beats) {
   290→    const pool = ACT_MOVEMENT_POOLS[beat.narrativeFunction];
   291→    let mv = beat.cameraMovement;
   292→    if (!pool.includes(mv)) mv = pool[0];
   293→    if (used.has(mv)) {
   294→      const alt = pool.find((k) => !used.has(k));
   295→      if (alt) mv = alt;
   296→    }
   297→    used.add(mv);
   298→    beat.cameraMovement = mv;
   299→  }
   300→  return beats;
   301→}

   303→async function stageScreenwriting(
   304→  runId: string,
   305→  project: Project
   306→): Promise<StageResult> {
   307→  const provider = getTextProvider();
   308→  const base = buildScreenwritingMessages(project);
   309→
   310→  let beats: StoryBeat[] | null = null;
   311→  let lastRaw = "";
   312→  for (let attempt = 0; attempt < 2 && !beats; attempt++) {
   313→    const { text } = await provider.complete({
   314→      messages: [
   315→        base.messages[0],
   316→        {
   317→          role: "user",
   318→          content:
   319→            attempt === 0
   320→              ? base.messages[1].content
   321→              : `${base.messages[1].content}\n\n注意：上一次输出无法解析。请严格只输出一个合法 JSON 对象（含 storyBeats 数组），不要任何其他文字。`,
   322→        },
   323→      ],
   324→      thinking: { type: project.thinkingLevel === "low" ? "disabled" : "enabled" },
   325→    });
   326→    lastRaw = text;
   327→    beats = parseBeats(text);
   328→  }
   329→  if (!beats) {
   330→    throw new Error(`剧本解析失败（${lastRaw.slice(0, 120)}…）`);
   331→  }
   332→  await updateProject(project.id, { storyBeats: beats, status: "beatsReady" });
   333→  return {
   334→    summary: `三幕剧本定稿：${beats
   335→      .map((b) => NARRATIVE_META[b.narrativeFunction].label)
   336→      .join(" → ")}，运镜 ${beats.map((b) => b.cameraMovement).join(" / ")}`,
   337→  };
   338→}

   350→function buildCastingMessages(beats: DbBeat[], project: Project): string {
   351→  const script = beats
   352→    .map(
   353→      (b) =>
   354→        `【第 ${b.index} 幕 · ${b.actTitle}】\n画面：${b.description}\n金句：${b.dialogueSubtitle}\n提示：${b.promptGuidance}`
   355→    )
   356→    .join("\n\n");
   357→
   358→  return `你是微短剧剧组的选角导演兼美术指导。任务：通读三幕剧本，建立全剧资产库（角色 / 场景 / 道具），供后续所有镜头做「一致性锚定」——每个镜头的提示词都将引用这里的定妆描述，保证跨镜头不换脸、不穿帮。
   359→
   360→【题材调性】${project.genreName}（${project.genreBadge}）；英文影调锚点：${project.tonePrompt}
   361→
   362→【三幕剧本】
   363→${script}
   364→
   365→【要求（严格遵守）】
   366→1. characters：主角、对手与关键配角，最多 6 个。lockedAppearance 为中文定妆描述（25~60 字：年龄、性别、身份气质、发型、脸型特征、服装、标志物）；visualAnchor 为对应英文串（15~30 words，如 "35-year-old Chinese male special forces captain, buzz cut, scar on left brow, black tactical vest"）
   367→2. scenes：关键场景，最多 5 个。name 为短名（如「宴会厅」「冷宫」）；lockedAppearance 描述空间与光线；visualAnchor 为英文
   368→3. props：推动剧情的关键道具，最多 4 个，没有则输出空数组
   369→4. 名称必须与剧本用词一致，不得杜撰剧本中未出现的角色
   370→5. visualAnchor 是所有镜头共用的视觉锚点，聚焦「跨镜头不变的外观特征」，不要写动作
   371→
   372→【输出格式】只输出 JSON，禁止解释文字、禁止 markdown 围栏：
   373→{"characters":[{"name":"…","lockedAppearance":"…","visualAnchor":"…"}],"scenes":[{"name":"…","lockedAppearance":"…","visualAnchor":"…"}],"props":[{"name":"…","lockedAppearance":"…","visualAnchor":"…"}]}`;
   374→}
   375→
   376→function parseCasting(
   377→  raw: string
   378→): { characters: CastingAssetDraft[]; scenes: CastingAssetDraft[]; props: CastingAssetDraft[] } | null {
   379→  const data = parseJsonLoose(raw) as Record<string, unknown> | null;
   380→  if (!data) return null;
   381→
   382→  const map = (v: unknown, max: number): CastingAssetDraft[] =>
   383→    (Array.isArray(v) ? v : [])
   384→      .map((item) => {
   385→        const o = item as Record<string, unknown>;
   386→        const name = typeof o?.name === "string" ? o.name.trim().slice(0, 20) : "";
   387→        if (!name) return null;
   388→        const lockedAppearance =
   389→          typeof o?.lockedAppearance === "string" ? o.lockedAppearance.trim().slice(0, 200) : "";
   390→        const visualAnchor =
   391→          typeof o?.visualAnchor === "string" ? o.visualAnchor.trim().slice(0, 300) : "";
   392→        if (!lockedAppearance && !visualAnchor) return null;
   393→        return {
   394→          name,
   395→          lockedAppearance: lockedAppearance || "（定妆描述待补充）",
   396→          visualAnchor: visualAnchor || lockedAppearance,
   397→        } satisfies CastingAssetDraft;
   398→      })
   399→      .filter((x): x is CastingAssetDraft => !!x)
   400→      .slice(0, max);
   401→
   402→  const characters = map(data.characters, 6);
   403→  const scenes = map(data.scenes, 5);
   404→  const props = map(data.props, 4);
   405→  if (characters.length === 0 && scenes.length === 0) return null;
   406→  return { characters, scenes, props };
   407→}

   428→        { role: "assistant", content: prompt },
   429→        {
   430→          role: "user",
   431→          content:
   432→            attempt === 0
   433→              ? "请建立资产库并只输出 JSON。"
   434→              : "上一次输出无法解析。请严格只输出一个合法 JSON 对象（characters/scenes/props 三个数组），不要任何其他文字。",
   435→        },
   436→      ],
   437→      thinking: { type: "disabled" },
   438→    });
   439→    lastRaw = text;
   440→    casting = parseCasting(text);
   441→  }
   442→  if (!casting) {
   443→    throw new Error(`选角定妆解析失败（${lastRaw.slice(0, 120)}…）`);
   444→  }
   445→
   446→  // 幂等 upsert：同名资产更新锚点（保证与最新剧本一致）
   447→  const kindGroups: Array<[AssetKind, CastingAssetDraft[]]> = [
   448→    ["character", casting.characters],
   449→    ["scene", casting.scenes],
   450→    ["prop", casting.props],
   451→  ];
   452→  let count = 0;
   453→  for (const [kind, drafts] of kindGroups) {
   454→    for (const draft of drafts) {
   455→      await db.asset.upsert({
   456→        where: {
   457→          projectId_kind_name: {
   458→            projectId: project.id,
   459→            kind,
   460→            name: draft.name,
   461→          },
   462→        },
   463→        create: {
   464→          projectId: project.id,
   465→          kind,
   466→          name: draft.name,
   467→          lockedAppearance: draft.lockedAppearance,
   468→          visualAnchor: draft.visualAnchor,
   469→          source: "crew",
   470→        },
   471→        update: {
   472→          lockedAppearance: draft.lockedAppearance,
   473→          visualAnchor: draft.visualAnchor,
   474→        },
   475→      });
   476→      count++;
   477→    }
   478→  }
   479→  await appendLog(
   480→    runId,
   481→    "casting",
   482→    `资产入库：角色 ${casting.characters.length} · 场景 ${casting.scenes.length} · 道具 ${casting.props.length}`
   483→  );
   484→  return {
   485→    summary: `资产库建立：${count} 项定妆锁定（角色 ${casting.characters.length} / 场景 ${casting.scenes.length} / 道具 ${casting.props.length}）`,
   486→  };
   487→}

   493→async function stageDirecting(
   494→  runId: string,
   495→  project: Project
   496→): Promise<StageResult> {
   497→  const [beats, assets] = await Promise.all([
   498→    db.storyBeat.findMany({
   499→      where: { projectId: project.id },
   500→      orderBy: { index: "asc" },
   501→    }),
   502→    db.asset.findMany({ where: { projectId: project.id } }),
   503→  ]);
   504→  if (beats.length < 3) {
   505→    throw new Error("剧本尚未就绪，请先完成编剧阶段");
   506→  }
   507→
   508→  const roster = {
   509→    characters: assets.filter((a) => a.kind === "character").map((a) => a.name),
   510→    scenes: assets.filter((a) => a.kind === "scene").map((a) => a.name),
   511→    props: assets.filter((a) => a.kind === "prop").map((a) => a.name),
   512→  };
   513→
   514→  let totalShots = 0;
   515→  for (const beat of beats) {
   516→    const shots = await breakdownAct({
   517→      genreName: project.genreName,
   518→      genreBadge: project.genreBadge,
   519→      tonePrompt: project.tonePrompt,
   520→      synopsis: project.synopsis,
   521→      beats: beats.map((b) => ({
   522→        index: b.index,
   523→        actTitle: b.actTitle,
   524→        narrativeFunction: b.narrativeFunction as StoryBeat["narrativeFunction"],
   525→        pacingNote: b.pacingNote,
   526→        description: b.description,
   527→        dialogueSubtitle: b.dialogueSubtitle,
   528→        cameraMovement: b.cameraMovement,
   529→        promptGuidance: b.promptGuidance,
   530→      })),
   531→      targetIndex: beat.index,
   532→      assetRoster: roster,
   533→    });
   534→    const clips = await replaceActClips(project.id, beat.index, shots);
   535→    await bindRefsFromShots(project.id, beat.index, shots, clips);
   536→    await updateBeatEnergy(project.id, beat.index, {
   537→      energyStart: shots[0]?.energy ?? null,
   538→      energyEnd: shots[shots.length - 1]?.energy ?? null,
   539→    });
   540→    totalShots += shots.length;
   541→    await appendLog(
   542→      runId,
   543→      "director",
   544→      `第 ${beat.index} 幕拆解完成：${shots.length} 镜（${shots
   545→        .map((s) => s.shotSize)
   546→        .join("→")}，能量 ${shots[0]?.energy ?? "-"}→${shots[shots.length - 1]?.energy ?? "-"}）`
   547→    );
   548→  }
   549→  return {
   550→    summary: `三幕拆镜完成：共 ${totalShots} 个镜头，正反打景别交替已就位`,
   551→  };
   552→}

   554→/** 按名称把拆镜产物中的资产引用解析为 ShotRef（精确名 → 包含式回退） */
   555→export async function bindRefsFromShots(
   556→  projectId: string,
   557→  beatIndex: number,
   558→  shots: ShotBreakdownItem[],
   559→  clips: Array<{ id: string; beatIndex: number; order: number }>
   560→): Promise<number> {
   561→  const assets = await db.asset.findMany({ where: { projectId } });
   562→  const byKind = (kind: AssetKind) => assets.filter((a) => a.kind === kind);
   563→  const resolve = (kind: AssetKind, name: string) => {
   564→    if (!name) return null;
   565→    const pool = byKind(kind);
   566→    return (
   567→      pool.find((a) => a.name === name) ??
   568→      pool.find(
   569→        (a) => a.name.includes(name) || name.includes(a.name)
   570→      ) ??
   571→      null
   572→    );
   573→  };
   574→
   575→  const actClips = clips
   576→    .filter((c) => c.beatIndex === beatIndex)
   577→    .sort((a, b) => a.order - b.order);
   578→  let bound = 0;
   579→  for (let i = 0; i < actClips.length && i < shots.length; i++) {
   580→    const shot = shots[i];
   581→    const clip = actClips[i];
   582→    const targets = new Set<string>();
   583→    for (const name of shot.characters) {
   584→      const asset = resolve("character", name);
   585→      if (asset) targets.add(asset.id);
   586→    }
   587→    const scene = resolve("scene", shot.scene);
   588→    if (scene) targets.add(scene.id);
   589→    for (const name of shot.props) {
   590→      const asset = resolve("prop", name);
   591→      if (asset) targets.add(asset.id);
   592→    }
   593→    for (const assetId of targets) {
   594→      await db.shotRef.upsert({
   595→        where: { shotId_assetId: { shotId: clip.id, assetId } },
   596→        create: { shotId: clip.id, assetId },
   597→        update: {},
   598→      });
   599→      bound++;
   600→    }
   601→  }
   602→  return bound;
   603→}

   609→export async function anchorProjectClips(
   610→  projectId: string,
   611→  onlyClipIds?: string[]
   612→): Promise<{ anchored: number; skipped: number }> {
   613→  const [project, clips, refs] = await Promise.all([
   614→    db.project.findUnique({ where: { id: projectId } }),
   615→    db.timelineClip.findMany({
   616→      where: { projectId, ...(onlyClipIds ? { id: { in: onlyClipIds } } : {}) },
   617→      orderBy: { order: "asc" },
   618→    }),
   619→    db.shotRef.findMany({
   620→      where: { shot: { projectId } },
   621→      include: { asset: true },
   622→    }),
   623→  ]);
   624→  if (!project) throw new RouteError("项目不存在或已被删除", 404);
   625→
   626→  const refsByClip = new Map<string, typeof refs>();
   627→  for (const r of refs) {
   628→    const list = refsByClip.get(r.shotId) ?? [];
   629→    list.push(r);
   630→    refsByClip.set(r.shotId, list);
   631→  }
   632→
   633→  let anchored = 0;
   634→  let skipped = 0;
   635→  for (const clip of clips) {
   636→    if (clip.promptVersion === "manual") {
   637→      skipped++;
   638→      continue;
   639→    }
   640→    const bound = refsByClip.get(clip.id) ?? [];
   641→    const characters = bound
   642→      .filter((r) => r.asset.kind === "character")
   643→      .map((r) => ({
   644→        name: r.asset.name,
   645→        visualAnchor: r.asset.visualAnchor,
   646→        stateNote: r.stateNote,
   647→      }));
   648→    const sceneRef = bound.find((r) => r.asset.kind === "scene");
   649→    const props = bound
   650→      .filter((r) => r.asset.kind === "prop")
   651→      .map((r) => ({
   652→        name: r.asset.name,
   653→        visualAnchor: r.asset.visualAnchor,
   654→        stateNote: r.stateNote,
   655→      }));
   656→
   657→    const { finalPrompt, fingerprint } = composeVideoPrompt({
   658→      shot: {
   659→        visualDesc: clip.visualDesc ?? clip.name,
   660→        visualPrompt: clip.visualPrompt,
   661→        cameraMovement: clip.cameraMovement,
   662→        shotSize: clip.shotSize,
   663→        subtitle: clip.subtitle,
   664→      },
   665→      characters,
   666→      scene: sceneRef
   667→        ? {
   668→            name: sceneRef.asset.name,
   669→            visualAnchor: sceneRef.asset.visualAnchor,
   670→            stateNote: sceneRef.stateNote,
   671→          }
   672→        : null,
   673→      props,
   674→      tonePrompt: project.tonePrompt,
   675→      aspectRatio: project.aspectRatio,
   676→      plannedMs: clip.plannedMs,
   677→      resolution: project.resolution,
   678→    });
   679→
   680→    await db.timelineClip.update({
   681→      where: { id: clip.id },
   682→      data: { finalPrompt, promptVersion: fingerprint },
   683→    });
   684→    anchored++;
   685→  }
   686→  return { anchored, skipped };
   687→}

   736→async function executeRun(runId: string): Promise<void> {
   737→  try {
   738→    const run = await db.productionRun.findUnique({ where: { id: runId } });
   739→    if (!run || run.status !== "running") return;
   740→
   741→    for (const def of RUN_STAGES) {
   742→      const current = await loadRun(runId);
   743→      if (current.status !== "running") return; // 被外部取消/覆盖
   744→      const stages = parseStages(current.stages);
   745→      const stage = stages.find((s) => s.key === def.key);
   746→      if (!stage || stage.status === "succeeded") continue;
   747→
   748→      // —— 阶段置 running ——
   749→      stage.status = "running";
   750→      stage.startedAt = new Date().toISOString();
   751→      stage.error = null;
   752→      await patchStages(runId, stages);
   753→      await db.productionRun.update({
   754→        where: { id: runId },
   755→        data: { stage: def.key },
   756→      });
   757→      const project = await db.project.findUnique({
   758→        where: { id: current.projectId },
   759→      });
   760→      if (!project) throw new Error("项目不存在或已被删除");
   761→
   762→      const startMsg =
   763→        STAGE_START_MESSAGE[def.key] ?? `进入「${def.name}」阶段`;
   764→      const modeLabel =
   765→        def.key === "screenwriting" ? thinkingModeLabel(project.thinkingLevel) : "";
   766→      await appendLog(
   767→        runId,
   768→        def.role,
   769→        startMsg.replace("{mode}", modeLabel)
   770→      );
   771→
   772→      try {
   773→        const result = await STAGE_FN[def.key](runId, project);
   774→        const fresh = parseStages((await loadRun(runId)).stages);
   775→        const stageRow = fresh.find((s) => s.key === def.key);
   776→        if (stageRow) {
   777→          stageRow.status = "succeeded";
   778→          stageRow.finishedAt = new Date().toISOString();
   779→          await patchStages(runId, fresh);
   780→        }
   781→        await appendLog(runId, def.role, result.summary);
   782→        await recomputeProgress(runId);
   783→      } catch (err) {
   784→        const message =
   785→          err instanceof Error ? err.message : "未知错误";
   786→        const fresh = parseStages((await loadRun(runId)).stages);
   787→        const stageRow = fresh.find((s) => s.key === def.key);
   788→        if (stageRow) {
   789→          stageRow.status = "failed";
   790→          stageRow.finishedAt = new Date().toISOString();
   791→          stageRow.error = message;
   792→          await patchStages(runId, fresh);
   793→        }
   794→        await db.productionRun.update({
   795→          where: { id: runId },
   796→          data: { status: "failed", error: `「${def.name}」阶段失败：${message}` },
   797→        });
   798→        await appendLog(runId, def.role, `阶段失败：${message}`);
   799→        return;
   800→      }
   801→    }
   802→
   803→    await db.productionRun.update({
   804→      where: { id: runId },
   805→      data: { status: "succeeded", progress: 100 },
   806→    });
   807→    await appendLog(
   808→      runId,
   809→      "producer",
   810→      "全片杀青 🎬 制作包已就绪：三幕剧本 · 分镜表 · 资产定妆 · 镜头级提示词 · 三轨 · 能量曲线，可导出交付"
   811→    );
   812→  } catch (err) {
   813→    const message = err instanceof Error ? err.message : "未知错误";
   814→    await db.productionRun
   815→      .update({
   816→        where: { id: runId },
   817→        data: { status: "failed", error: message },
   818→      })
   819→      .catch(() => undefined);
   820→  }
   821→}

   827→export async function startRun(projectId: string): Promise<RunDTO> {
   828→  const project = await db.project.findUnique({ where: { id: projectId } });
   829→  if (!project) throw new RouteError("项目不存在或已被删除", 404);

   859→/** 断点续跑：已成功阶段保留，从首个未成功阶段继续 */
   860→export async function retryRun(runId: string): Promise<RunDTO> {
   861→  const run = await loadRun(runId);
   862→  if (run.status === "running") {
   863→    throw new RouteError("流水线正在运行中，无需重试", 409);
   864→  }
   865→  const stages = parseStages(run.stages).map((s) =>
   866→    s.status === "failed"
   867→      ? { ...s, status: "pending", error: null, startedAt: null, finishedAt: null }
   868→      : s
   869→  );
   870→  const resume = stages.find((s) => s.status !== "succeeded");
   871→  await db.productionRun.update({
   872→    where: { id: runId },
   873→    data: {
   874→      status: "running",
   875→      error: null,
   876→      stages: JSON.stringify(stages),
   877→    },
   878→  });
   879→  await appendLog(
   880→    runId,
   881→    "producer",
   882→    resume
   883→      ? `断点续拍：从「${resume.name}」阶段继续（已完成阶段不重复执行）`
   884→      : "复核完成，直接进入收尾"
   885→  );
   886→  void executeRun(runId);
   887→  const fresh = await loadRun(runId);
   888→  return toRunDTO(fresh);
   889→}
