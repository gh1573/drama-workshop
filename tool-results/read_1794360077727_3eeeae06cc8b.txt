    59→function parseStages(raw: string): RunStageInfo[] {
    60→  try {
    61→    const parsed = JSON.parse(raw) as RunStageInfo[];
    62→    if (Array.isArray(parsed) && parsed.length === RUN_STAGES.length) {
    63→      return parsed;
    64→    }
    65→  } catch {
    66→    /* fallthrough */
    67→  }
    68→  return freshStages();
    69→}

    72→  try {
    73→    const parsed = JSON.parse(raw) as RunLogEntry[];
    74→    if (Array.isArray(parsed)) return parsed;
    75→  } catch {
    76→    /* fallthrough */
    77→  }
    78→  return [];
    79→}

   147→async function recomputeProgress(runId: string): Promise<void> {
   148→  const row = await loadRun(runId);
   149→  const stages = parseStages(row.stages);
   150→  const done = stages
   151→    .filter((s) => s.status === "succeeded")
   152→    .reduce(
   153→      (sum, s) =>
   154→        sum + (RUN_STAGES.find((d) => d.key === s.key)?.weight ?? 0),
   155→      0
   156→    );
   157→  const total = RUN_STAGES.reduce((sum, d) => sum + d.weight, 0);
   158→  await db.productionRun.update({
   159→    where: { id: runId },
   160→    data: { progress: Math.min(100, Math.round((done / total) * 100)) },
   161→  });
   162→}

   255→function parseBeats(raw: string): StoryBeat[] | null {
   256→  const data = parseJsonLoose(raw) as { storyBeats?: unknown[] } | null;
   257→  const arr = data?.storyBeats;
   258→  if (!Array.isArray(arr) || arr.length < 3) return null;
   259→
   260→  const funcs: NarrativeFunction[] = ["hook", "climax", "cliffhanger"];
   261→  const beats: StoryBeat[] = [];
   262→  for (let i = 0; i < 3; i++) {
   263→    const item = arr[i] as Record<string, unknown>;
   264→    const fn = funcs.includes(item?.narrativeFunction as NarrativeFunction)
   265→      ? (item.narrativeFunction as NarrativeFunction)
   266→      : funcs[i];
   267→    const meta = NARRATIVE_META[fn];
   268→    const str = (v: unknown, fallback: string) =>
   269→      typeof v === "string" && v.trim() ? v.trim() : fallback;
   270→    const cam =
   271→      typeof item?.cameraMovement === "string" &&
   272→      item.cameraMovement in CAMERA_MOVEMENTS
   273→        ? item.cameraMovement
   274→        : "";
   275→    beats.push({
   276→      index: i + 1,
   277→      actTitle: `${meta.actNo}：${meta.label} (${meta.tag})`,
   278→      narrativeFunction: fn,
   279→      pacingNote: str(item?.pacingNote, PACING_FALLBACK[fn]),
   280→      description: str(item?.description, "（分镜内容生成异常，请重试）"),
   281→      dialogueSubtitle: str(item?.dialogueSubtitle, "台词生成异常，请重试"),
   282→      cameraMovement: cam,
   283→      promptGuidance: str(item?.promptGuidance, "提示生成异常，请重试"),
   284→      energyStart: null,
   285→      energyEnd: null,
   286→    });
   287→  }
   288→  // 运镜合规修复：池外 → 本幕池；三幕互斥 → 换池内备选
   289→  const used = new Set<string>();
   290→  for (const beat of beats) {
   291→    const pool = ACT_MOVEMENT_POOLS[beat.narrativeFunction];
   292→    let mv = beat.cameraMovement;
   293→    if (!pool.includes(mv)) mv = pool[0];
   294→    if (used.has(mv)) {
   295→      const alt = pool.find((k) => !used.has(k));
   296→      if (alt) mv = alt;
   297→    }
   298→    used.add(mv);
   299→    beat.cameraMovement = mv;
   300→  }
   301→  return beats;
   302→}

   304→async function stageScreenwriting(
   305→  runId: string,
   306→  project: Project
   307→): Promise<StageResult> {
   308→  const provider = getTextProvider();
   309→  const base = buildScreenwritingMessages(project);
   310→
   311→  let beats: StoryBeat[] | null = null;
   312→  let lastRaw = "";
   313→  for (let attempt = 0; attempt < 2 && !beats; attempt++) {
   314→    const { text } = await provider.complete({
   315→      messages: [
   316→        base.messages[0],
   317→        {
   318→          role: "user",
   319→          content:
   320→            attempt === 0
   321→              ? base.messages[1].content
   322→              : `${base.messages[1].content}\n\n注意：上一次输出无法解析。请严格只输出一个合法 JSON 对象（含 storyBeats 数组），不要任何其他文字。`,
   323→        },
   324→      ],
   325→      thinking: { type: project.thinkingLevel === "low" ? "disabled" : "enabled" },
   326→    });
   327→    lastRaw = text;
   328→    beats = parseBeats(text);
   329→  }
   330→  if (!beats) {
   331→    throw new Error(`剧本解析失败（${lastRaw.slice(0, 120)}…）`);
   332→  }
   333→  await updateProject(project.id, { storyBeats: beats, status: "beatsReady" });
   334→  return {
   335→    summary: `三幕剧本定稿：${beats
   336→      .map((b) => NARRATIVE_META[b.narrativeFunction].label)
   337→      .join(" → ")}，运镜 ${beats.map((b) => b.cameraMovement).join(" / ")}`,
   338→  };
   339→}

   351→function buildCastingMessages(beats: DbBeat[], project: Project): string {
   352→  const script = beats
   353→    .map(
   354→      (b) =>
   355→        `【第 ${b.index} 幕 · ${b.actTitle}】\n画面：${b.description}\n金句：${b.dialogueSubtitle}\n提示：${b.promptGuidance}`
   356→    )

   359→  return `你是微短剧剧组的选角导演兼美术指导。任务：通读三幕剧本，建立全剧资产库（角色 / 场景 / 道具），供后续所有镜头做「一致性锚定」——每个镜头的提示词都将引用这里的定妆描述，保证跨镜头不换脸、不穿帮。
   360→
   361→【题材调性】${project.genreName}（${project.genreBadge}）；英文影调锚点：${project.tonePrompt}
   362→
   363→【三幕剧本】
   364→${script}
   365→
   366→【要求（严格遵守）】
   367→1. characters：主角、对手与关键配角，最多 6 个。lockedAppearance 为中文定妆描述（25~60 字：年龄、性别、身份气质、发型、脸型特征、服装、标志物）；visualAnchor 为对应英文串（15~30 words，如 "35-year-old Chinese male special forces captain, buzz cut, scar on left brow, black tactical vest"）
   368→2. scenes：关键场景，最多 5 个。name 为短名（如「宴会厅」「冷宫」）；lockedAppearance 描述空间与光线；visualAnchor 为英文
   369→3. props：推动剧情的关键道具，最多 4 个，没有则输出空数组
   370→4. 名称必须与剧本用词一致，不得杜撰剧本中未出现的角色
   371→5. visualAnchor 是所有镜头共用的视觉锚点，聚焦「跨镜头不变的外观特征」，不要写动作
   372→
   373→【输出格式】只输出 JSON，禁止解释文字、禁止 markdown 围栏：
   374→{"characters":[{"name":"…","lockedAppearance":"…","visualAnchor":"…"}],"scenes":[{"name":"…","lockedAppearance":"…","visualAnchor":"…"}],"props":[{"name":"…","lockedAppearance":"…","visualAnchor":"…"}]}`;
   375→}
   376→
   377→function parseCasting(
   378→  raw: string
   379→): { characters: CastingAssetDraft[]; scenes: CastingAssetDraft[]; props: CastingAssetDraft[] } | null {
   380→  const data = parseJsonLoose(raw) as Record<string, unknown> | null;
   381→  if (!data) return null;
   382→
   383→  const map = (v: unknown, max: number): CastingAssetDraft[] =>
   384→    (Array.isArray(v) ? v : [])
   385→      .map((item) => {
   386→        const o = item as Record<string, unknown>;
   387→        const name = typeof o?.name === "string" ? o.name.trim().slice(0, 20) : "";
   388→        if (!name) return null;
   389→        const lockedAppearance =
   390→          typeof o?.lockedAppearance === "string" ? o.lockedAppearance.trim().slice(0, 200) : "";
   391→        const visualAnchor =
   392→          typeof o?.visualAnchor === "string" ? o.visualAnchor.trim().slice(0, 300) : "";
   393→        if (!lockedAppearance && !visualAnchor) return null;
   394→        return {
   395→          name,
   396→          lockedAppearance: lockedAppearance || "（定妆描述待补充）",
   397→          visualAnchor: visualAnchor || lockedAppearance,
   398→        } satisfies CastingAssetDraft;
   399→      })
   400→      .filter((x): x is CastingAssetDraft => !!x)
   401→      .slice(0, max);
   402→
   403→  const characters = map(data.characters, 6);
   404→  const scenes = map(data.scenes, 5);
   405→  const props = map(data.props, 4);
   406→  if (characters.length === 0 && scenes.length === 0) return null;
   407→  return { characters, scenes, props };
   408→}

   427→    const { text } = await provider.complete({
   428→      messages: [
   429→        { role: "assistant", content: prompt },
   430→        {
   431→          role: "user",
   432→          content:
   433→            attempt === 0
   434→              ? "请建立资产库并只输出 JSON。"
   435→              : "上一次输出无法解析。请严格只输出一个合法 JSON 对象（characters/scenes/props 三个数组），不要任何其他文字。",
   436→        },
   437→      ],
   438→      thinking: { type: "disabled" },
   439→    });
   440→    lastRaw = text;
   441→    casting = parseCasting(text);
   442→  }
   443→  if (!casting) {
   444→    throw new Error(`选角定妆解析失败（${lastRaw.slice(0, 120)}…）`);
   445→  }
   446→
   447→  // 幂等 upsert：同名资产更新锚点（保证与最新剧本一致）
   448→  const kindGroups: Array<[AssetKind, CastingAssetDraft[]]> = [
   449→    ["character", casting.characters],
   450→    ["scene", casting.scenes],
   451→    ["prop", casting.props],
   452→  ];
   453→  let count = 0;
   454→  for (const [kind, drafts] of kindGroups) {
   455→    for (const draft of drafts) {
   456→      await db.asset.upsert({
   457→        where: {
   458→          projectId_kind_name: {
   459→            projectId: project.id,
   460→            kind,
   461→            name: draft.name,
   462→          },
   463→        },
   464→        create: {
   465→          projectId: project.id,
   466→          kind,
   467→          name: draft.name,
   468→          lockedAppearance: draft.lockedAppearance,
   469→          visualAnchor: draft.visualAnchor,
   470→          source: "crew",
   471→        },
   472→        update: {
   473→          lockedAppearance: draft.lockedAppearance,
   474→          visualAnchor: draft.visualAnchor,
   475→        },
   476→      });
   477→      count++;
   478→    }
   479→  }
   480→  await appendLog(
   481→    runId,
   482→    "casting",
   483→    `资产入库：角色 ${casting.characters.length} · 场景 ${casting.scenes.length} · 道具 ${casting.props.length}`
   484→  );
   485→  return {
   486→    summary: `资产库建立：${count} 项定妆锁定（角色 ${casting.characters.length} / 场景 ${casting.scenes.length} / 道具 ${casting.props.length}）`,
   487→  };

   494→async function stageDirecting(
   495→  runId: string,
   496→  project: Project
   497→): Promise<StageResult> {
   498→  const [beats, assets] = await Promise.all([
   499→    db.storyBeat.findMany({
   500→      where: { projectId: project.id },
   501→      orderBy: { index: "asc" },
   502→    }),
   503→    db.asset.findMany({ where: { projectId: project.id } }),
   504→  ]);
   505→  if (beats.length < 3) {
   506→    throw new Error("剧本尚未就绪，请先完成编剧阶段");
   507→  }
   508→
   509→  const roster = {
   510→    characters: assets.filter((a) => a.kind === "character").map((a) => a.name),
   511→    scenes: assets.filter((a) => a.kind === "scene").map((a) => a.name),
   512→    props: assets.filter((a) => a.kind === "prop").map((a) => a.name),
   513→  };
   514→
   515→  let totalShots = 0;
   516→  let skippedActs = 0;
   517→  for (const beat of beats) {
   518→    // 成片保护：该幕已有已生成镜头时跳过拆解（不覆盖劳动成果），
   519→    // 由日志明确告知；如需重拍请先在镜头轨把该幕镜头重置为待生成
   520→    const hasReadyClip = await db.timelineClip.findFirst({
   521→      where: { projectId: project.id, beatIndex: beat.index, status: "ready" },
   522→      select: { id: true },
   523→    });
   524→    if (hasReadyClip) {
   525→      skippedActs++;
   526→      await appendLog(
   527→        runId,
   528→        "director",
   529→        `第 ${beat.index} 幕已有已生成镜头，跳过拆解以保护成片（重拍请先重置该幕镜头）`
   530→      );
   531→      continue;
   532→    }
   533→    const shots = await breakdownAct({
   534→      genreName: project.genreName,
   535→      genreBadge: project.genreBadge,
   536→      tonePrompt: project.tonePrompt,
   537→      synopsis: project.synopsis,
   538→      beats: beats.map((b) => ({
   539→        index: b.index,
   540→        actTitle: b.actTitle,
   541→        narrativeFunction: b.narrativeFunction as StoryBeat["narrativeFunction"],
   542→        pacingNote: b.pacingNote,
   543→        description: b.description,
   544→        dialogueSubtitle: b.dialogueSubtitle,
   545→        cameraMovement: b.cameraMovement,
   546→        promptGuidance: b.promptGuidance,
   547→      })),
   548→      targetIndex: beat.index,
   549→      assetRoster: roster,
   550→    });
   551→    const clips = await replaceActClips(project.id, beat.index, shots);
   552→    await bindRefsFromShots(project.id, beat.index, shots, clips);
   553→    await updateBeatEnergy(project.id, beat.index, {
   554→      energyStart: shots[0]?.energy ?? null,
   555→      energyEnd: shots[shots.length - 1]?.energy ?? null,
   556→    });
   557→    totalShots += shots.length;
   558→    await appendLog(
   559→      runId,
   560→      "director",
   561→      `第 ${beat.index} 幕拆解完成：${shots.length} 镜（${shots
   562→        .map((s) => s.shotSize)
   563→        .join("→")}，能量 ${shots[0]?.energy ?? "-"}→${shots[shots.length - 1]?.energy ?? "-"}）`
   564→    );
   565→  }
   566→  return {
   567→    summary: `三幕拆镜完成：共 ${totalShots} 个镜头，正反打景别交替已就位${
   568→      skippedActs > 0 ? `；${skippedActs} 幕含成片已跳过` : ""
   569→    }`,
   570→  };
   571→}

   573→/** 按名称把拆镜产物中的资产引用解析为 ShotRef（精确名 → 包含式回退） */
   574→export async function bindRefsFromShots(
   575→  projectId: string,
   576→  beatIndex: number,
   577→  shots: ShotBreakdownItem[],
   578→  clips: Array<{ id: string; beatIndex: number; order: number }>
   579→): Promise<number> {
   580→  const assets = await db.asset.findMany({ where: { projectId } });
   581→  const byKind = (kind: AssetKind) => assets.filter((a) => a.kind === kind);
   582→  const resolve = (kind: AssetKind, name: string) => {
   583→    if (!name) return null;
   584→    const pool = byKind(kind);
   585→    return (
   586→      pool.find((a) => a.name === name) ??
   587→      pool.find(
   588→        (a) => a.name.includes(name) || name.includes(a.name)
   589→      ) ??
   590→      null
   591→    );
   592→  };
   593→
   594→  const actClips = clips
   595→    .filter((c) => c.beatIndex === beatIndex)
   596→    .sort((a, b) => a.order - b.order);
   597→  let bound = 0;
   598→  for (let i = 0; i < actClips.length && i < shots.length; i++) {
   599→    const shot = shots[i];
   600→    const clip = actClips[i];
   601→    const targets = new Set<string>();
   602→    for (const name of shot.characters) {
   603→      const asset = resolve("character", name);
   604→      if (asset) targets.add(asset.id);
   605→    }
   606→    const scene = resolve("scene", shot.scene);
   607→    if (scene) targets.add(scene.id);
   608→    for (const name of shot.props) {
   609→      const asset = resolve("prop", name);
   610→      if (asset) targets.add(asset.id);
   611→    }
   612→    for (const assetId of targets) {
   613→      await db.shotRef.upsert({
   614→        where: { shotId_assetId: { shotId: clip.id, assetId } },
   615→        create: { shotId: clip.id, assetId },
   616→        update: {},
   617→      });
   618→      bound++;
   619→    }
   620→  }
   621→  return bound;
   622→}

   628→export async function anchorProjectClips(
   629→  projectId: string,
   630→  onlyClipIds?: string[]
   631→): Promise<{ anchored: number; skipped: number }> {
   632→  const [project, clips, refs] = await Promise.all([
   633→    db.project.findUnique({ where: { id: projectId } }),
   634→    db.timelineClip.findMany({
   635→      where: { projectId, ...(onlyClipIds ? { id: { in: onlyClipIds } } : {}) },
   636→      orderBy: { order: "asc" },
   637→    }),
   638→    db.shotRef.findMany({
   639→      where: { shot: { projectId } },
   640→      include: { asset: true },
   641→    }),
   642→  ]);
   643→  if (!project) throw new RouteError("项目不存在或已被删除", 404);
   644→
   645→  const refsByClip = new Map<string, typeof refs>();
   646→  for (const r of refs) {
   647→    const list = refsByClip.get(r.shotId) ?? [];
   648→    list.push(r);
   649→    refsByClip.set(r.shotId, list);
   650→  }
   651→
   652→  let anchored = 0;
   653→  let skipped = 0;
   654→  for (const clip of clips) {
   655→    if (clip.promptVersion === "manual") {
   656→      skipped++;
   657→      continue;
   658→    }
   659→    const bound = refsByClip.get(clip.id) ?? [];
   660→    const characters = bound
   661→      .filter((r) => r.asset.kind === "character")
   662→      .map((r) => ({
   663→        name: r.asset.name,
   664→        visualAnchor: r.asset.visualAnchor,
   665→        stateNote: r.stateNote,
   666→      }));
   667→    const sceneRef = bound.find((r) => r.asset.kind === "scene");
   668→    const props = bound
   669→      .filter((r) => r.asset.kind === "prop")
   670→      .map((r) => ({
   671→        name: r.asset.name,
   672→        visualAnchor: r.asset.visualAnchor,
   673→        stateNote: r.stateNote,
   674→      }));
   675→
   676→    const { finalPrompt, fingerprint } = composeVideoPrompt({
   677→      shot: {
   678→        visualDesc: clip.visualDesc ?? clip.name,
   679→        visualPrompt: clip.visualPrompt,
   680→        cameraMovement: clip.cameraMovement,
   681→        shotSize: clip.shotSize,
   682→        subtitle: clip.subtitle,
   683→      },
   684→      characters,
   685→      scene: sceneRef
   686→        ? {
   687→            name: sceneRef.asset.name,
   688→            visualAnchor: sceneRef.asset.visualAnchor,
   689→            stateNote: sceneRef.stateNote,
   690→          }
   691→        : null,
   692→      props,
   693→      tonePrompt: project.tonePrompt,
   694→      aspectRatio: project.aspectRatio,
   695→      plannedMs: clip.plannedMs,
   696→      resolution: project.resolution,
   697→    });
   698→
   699→    await db.timelineClip.update({
   700→      where: { id: clip.id },
   701→      data: { finalPrompt, promptVersion: fingerprint },
   702→    });
   703→    anchored++;
   704→  }
   705→  return { anchored, skipped };
   706→}

   755→async function executeRun(runId: string): Promise<void> {
   756→  try {
   757→    const run = await db.productionRun.findUnique({ where: { id: runId } });
   758→    if (!run || run.status !== "running") return;
   759→
   760→    for (const def of RUN_STAGES) {
   761→      const current = await loadRun(runId);
   762→      if (current.status !== "running") return; // 被外部取消/覆盖
   763→      const stages = parseStages(current.stages);
   764→      const stage = stages.find((s) => s.key === def.key);
   765→      if (!stage || stage.status === "succeeded") continue;
   766→
   767→      // —— 阶段置 running ——
   768→      stage.status = "running";
   769→      stage.startedAt = new Date().toISOString();
   770→      stage.error = null;
   771→      await patchStages(runId, stages);
   772→      await db.productionRun.update({
   773→        where: { id: runId },
   774→        data: { stage: def.key },
   775→      });
   776→      const project = await db.project.findUnique({
   777→        where: { id: current.projectId },
   778→      });
   779→      if (!project) throw new Error("项目不存在或已被删除");
   780→
   781→      const startMsg =
   782→        STAGE_START_MESSAGE[def.key] ?? `进入「${def.name}」阶段`;
   783→      const modeLabel =
   784→        def.key === "screenwriting" ? thinkingModeLabel(project.thinkingLevel) : "";
   785→      await appendLog(
   786→        runId,
   787→        def.role,
   788→        startMsg.replace("{mode}", modeLabel)
   789→      );
   790→
   791→      try {
   792→        const result = await STAGE_FN[def.key](runId, project);
   793→        const fresh = parseStages((await loadRun(runId)).stages);
   794→        const stageRow = fresh.find((s) => s.key === def.key);
   795→        if (stageRow) {
   796→          stageRow.status = "succeeded";
   797→          stageRow.finishedAt = new Date().toISOString();
   798→          await patchStages(runId, fresh);
   799→        }
   800→        await appendLog(runId, def.role, result.summary);
   801→        await recomputeProgress(runId);
   802→      } catch (err) {
   803→        const message =
   804→          err instanceof Error ? err.message : "未知错误";
   805→        const fresh = parseStages((await loadRun(runId)).stages);
   806→        const stageRow = fresh.find((s) => s.key === def.key);
   807→        if (stageRow) {
   808→          stageRow.status = "failed";
   809→          stageRow.finishedAt = new Date().toISOString();
   810→          stageRow.error = message;
   811→          await patchStages(runId, fresh);
   812→        }
   813→        await db.productionRun.update({
   814→          where: { id: runId },
   815→          data: { status: "failed", error: `「${def.name}」阶段失败：${message}` },
   816→        });
   817→        await appendLog(runId, def.role, `阶段失败：${message}`);
   818→        return;
   819→      }
   820→    }
   821→
   822→    await db.productionRun.update({
   823→      where: { id: runId },
   824→      data: { status: "succeeded", progress: 100 },
   825→    });
   826→    await appendLog(
   827→      runId,
   828→      "producer",
   829→      "全片杀青 🎬 制作包已就绪：三幕剧本 · 分镜表 · 资产定妆 · 镜头级提示词 · 三轨 · 能量曲线，可导出交付"
   830→    );
   831→  } catch (err) {
   832→    const message = err instanceof Error ? err.message : "未知错误";
   833→    await db.productionRun
   834→      .update({
   835→        where: { id: runId },
   836→        data: { status: "failed", error: message },
   837→      })
   838→      .catch(() => undefined);
   839→  }
   840→}

   878→/** 断点续跑：已成功阶段保留，从首个未成功阶段继续 */
   879→export async function retryRun(runId: string): Promise<RunDTO> {
   880→  const run = await loadRun(runId);
   881→  if (run.status === "running") {
   882→    throw new RouteError("流水线正在运行中，无需重试", 409);
   883→  }
   884→  const stages = parseStages(run.stages).map((s) =>
   885→    s.status === "failed"
   886→      ? { ...s, status: "pending", error: null, startedAt: null, finishedAt: null }
   887→      : s
   888→  );
   889→  const resume = stages.find((s) => s.status !== "succeeded");
   890→  await db.productionRun.update({
   891→    where: { id: runId },
   892→    data: {
   893→      status: "running",
   894→      error: null,
   895→      stages: JSON.stringify(stages),
   896→    },
   897→  });
   898→  await appendLog(
   899→    runId,
   900→    "producer",
   901→    resume
   902→      ? `断点续拍：从「${resume.name}」阶段继续（已完成阶段不重复执行）`
   903→      : "复核完成，直接进入收尾"
   904→  );
   905→  void executeRun(runId);
   906→  const fresh = await loadRun(runId);
   907→  return toRunDTO(fresh);
   908→}
