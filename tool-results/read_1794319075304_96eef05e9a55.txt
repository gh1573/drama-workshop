   146→   146→async function recomputeProgress(runId: string): Promise<void> {
   147→   147→  const row = await loadRun(runId);
   148→   148→  const stages = parseStages(row.stages);
   149→   149→  const done = stages
   150→   150→    .filter((s) => s.status === "succeeded")
   151→   151→    .reduce(
   152→   152→      (sum, s) =>
   153→   153→        sum + (RUN_STAGES.find((d) => d.key === s.key)?.weight ?? 0),
   154→   154→      0
   155→   155→    );
   156→   156→  const total = RUN_STAGES.reduce((sum, d) => sum + d.weight, 0);
   157→   157→  await db.productionRun.update({
   158→   158→    where: { id: runId },
   159→   159→    data: { progress: Math.min(100, Math.round((done / total) * 100)) },
   160→   160→  });
   161→   161→}

   254→   254→function parseBeats(raw: string): StoryBeat[] | null {
   255→   255→  const data = parseJsonLoose(raw) as { storyBeats?: unknown[] } | null;
   256→   256→  const arr = data?.storyBeats;
   257→   257→  if (!Array.isArray(arr) || arr.length < 3) return null;
   258→   258→
   259→   259→  const funcs: NarrativeFunction[] = ["hook", "climax", "cliffhanger"];
   260→   260→  const beats: StoryBeat[] = [];
   261→   261→  for (let i = 0; i < 3; i++) {
   262→   262→    const item = arr[i] as Record<string, unknown>;
   263→   263→    const fn = funcs.includes(item?.narrativeFunction as NarrativeFunction)
   264→   264→      ? (item.narrativeFunction as NarrativeFunction)
   265→   265→      : funcs[i];
   266→   266→    const meta = NARRATIVE_META[fn];
   267→   267→    const str = (v: unknown, fallback: string) =>
   268→   268→      typeof v === "string" && v.trim() ? v.trim() : fallback;
   269→   269→    const cam =
   270→   270→      typeof item?.cameraMovement === "string" &&
   271→   271→      item.cameraMovement in CAMERA_MOVEMENTS
   272→   272→        ? item.cameraMovement
   273→   273→        : "";
   274→   274→    beats.push({
   275→   275→      index: i + 1,
   276→   276→      actTitle: `${meta.actNo}：${meta.label} (${meta.tag})`,
   277→   277→      narrativeFunction: fn,
   278→   278→      pacingNote: str(item?.pacingNote, PACING_FALLBACK[fn]),
   279→   279→      description: str(item?.description, "（分镜内容生成异常，请重试）"),
   280→   280→      dialogueSubtitle: str(item?.dialogueSubtitle, "台词生成异常，请重试"),
   281→   281→      cameraMovement: cam,
   282→   282→      promptGuidance: str(item?.promptGuidance, "提示生成异常，请重试"),
   283→   283→      energyStart: null,
   284→   284→      energyEnd: null,
   285→   285→    });
   286→   286→  }
   287→   287→  // 运镜合规修复：池外 → 本幕池；三幕互斥 → 换池内备选
   288→   288→  const used = new Set<string>();
   289→   289→  for (const beat of beats) {
   290→   290→    const pool = ACT_MOVEMENT_POOLS[beat.narrativeFunction];
   291→   291→    let mv = beat.cameraMovement;
   292→   292→    if (!pool.includes(mv)) mv = pool[0];
   293→   293→    if (used.has(mv)) {
   294→   294→      const alt = pool.find((k) => !used.has(k));
   295→   295→      if (alt) mv = alt;
   296→   296→    }
   297→   297→    used.add(mv);
   298→   298→    beat.cameraMovement = mv;
   299→   299→  }
   300→   300→  return beats;
   301→   301→}

   366→   366→1. characters：主角、对手与关键配角，最多 6 个。lockedAppearance 为中文定妆描述（25~60 字：年龄、性别、身份气质、发型、脸型特征、服装、标志物）；visualAnchor 为对应英文串（15~30 words，如 "35-year-old Chinese male special forces captain, buzz cut, scar on left brow, black tactical vest"）
   367→   367→2. scenes：关键场景，最多 5 个。name 为短名（如「宴会厅」「冷宫」）；lockedAppearance 描述空间与光线；visualAnchor 为英文
   368→   368→3. props：推动剧情的关键道具，最多 4 个，没有则输出空数组
   369→   369→4. 名称必须与剧本用词一致，不得杜撰剧本中未出现的角色
   370→   370→5. visualAnchor 是所有镜头共用的视觉锚点，聚焦「跨镜头不变的外观特征」，不要写动作
   371→   371→
   372→   372→【输出格式】只输出 JSON，禁止解释文字、禁止 markdown 围栏：
   373→   373→{"characters":[{"name":"…","lockedAppearance":"…","visualAnchor":"…"}],"scenes":[{"name":"…","lockedAppearance":"…","visualAnchor":"…"}],"props":[{"name":"…","lockedAppearance":"…","visualAnchor":"…"}]}`;
   374→   374→}
   375→   375→
   376→   376→function parseCasting(
   377→   377→  raw: string
   378→   378→): { characters: CastingAssetDraft[]; scenes: CastingAssetDraft[]; props: CastingAssetDraft[] } | null {
   379→   379→  const data = parseJsonLoose(raw) as Record<string, unknown> | null;
   380→   380→  if (!data) return null;
   381→   381→
   382→   382→  const map = (v: unknown, max: number): CastingAssetDraft[] =>
   383→   383→    (Array.isArray(v) ? v : [])
   384→   384→      .map((item) => {
   385→   385→        const o = item as Record<string, unknown>;
   386→   386→        const name = typeof o?.name === "string" ? o.name.trim().slice(0, 20) : "";
   387→   387→        if (!name) return null;
   388→   388→        const lockedAppearance =
   389→   389→          typeof o?.lockedAppearance === "string" ? o.lockedAppearance.trim().slice(0, 200) : "";
   390→   390→        const visualAnchor =
   391→   391→          typeof o?.visualAnchor === "string" ? o.visualAnchor.trim().slice(0, 300) : "";
   392→   392→        if (!lockedAppearance && !visualAnchor) return null;
   393→   393→        return {
   394→   394→          name,
   395→   395→          lockedAppearance: lockedAppearance || "（定妆描述待补充）",
   396→   396→          visualAnchor: visualAnchor || lockedAppearance,
   397→   397→        } satisfies CastingAssetDraft;
   398→   398→      })
   399→   399→      .filter((x): x is CastingAssetDraft => !!x)
   400→   400→      .slice(0, max);
   401→   401→
   402→   402→  const characters = map(data.characters, 6);
   403→   403→  const scenes = map(data.scenes, 5);
   404→   404→  const props = map(data.props, 4);
   405→   405→  if (characters.length === 0 && scenes.length === 0) return null;
   406→   406→  return { characters, scenes, props };
   407→   407→}

   554→   554→/** 按名称把拆镜产物中的资产引用解析为 ShotRef（精确名 → 包含式回退） */
   555→   555→export async function bindRefsFromShots(
   556→   556→  projectId: string,
   557→   557→  beatIndex: number,
   558→   558→  shots: ShotBreakdownItem[],
   559→   559→  clips: Array<{ id: string; beatIndex: number; order: number }>
   560→   560→): Promise<number> {
   561→   561→  const assets = await db.asset.findMany({ where: { projectId } });
   562→   562→  const byKind = (kind: AssetKind) => assets.filter((a) => a.kind === kind);
   563→   563→  const resolve = (kind: AssetKind, name: string) => {
   564→   564→    if (!name) return null;
   565→   565→    const pool = byKind(kind);
   566→   566→    return (
   567→   567→      pool.find((a) => a.name === name) ??
   568→   568→      pool.find(
   569→   569→        (a) => a.name.includes(name) || name.includes(a.name)
   570→   570→      ) ??
   571→   571→      null
   572→   572→    );
   573→   573→  };
   574→   574→
   575→   575→  const actClips = clips
   576→   576→    .filter((c) => c.beatIndex === beatIndex)
   577→   577→    .sort((a, b) => a.order - b.order);
   578→   578→  let bound = 0;
   579→   579→  for (let i = 0; i < actClips.length && i < shots.length; i++) {
   580→   580→    const shot = shots[i];
   581→   581→    const clip = actClips[i];
   582→   582→    const targets = new Set<string>();
   583→   583→    for (const name of shot.characters) {
   584→   584→      const asset = resolve("character", name);
   585→   585→      if (asset) targets.add(asset.id);
   586→   586→    }
   587→   587→    const scene = resolve("scene", shot.scene);
   588→   588→    if (scene) targets.add(scene.id);
   589→   589→    for (const name of shot.props) {
   590→   590→      const asset = resolve("prop", name);
   591→   591→      if (asset) targets.add(asset.id);
   592→   592→    }
   593→   593→    for (const assetId of targets) {
   594→   594→      await db.shotRef.upsert({
   595→   595→        where: { shotId_assetId: { shotId: clip.id, assetId } },
   596→   596→        create: { shotId: clip.id, assetId },
   597→   597→        update: {},
   598→   598→      });
   599→   599→      bound++;
   600→   600→    }
   601→   601→  }
   602→   602→  return bound;
   603→   603→}

   609→   609→export async function anchorProjectClips(
   610→   610→  projectId: string,
   611→   611→  onlyClipIds?: string[]
   612→   612→): Promise<{ anchored: number; skipped: number }> {
   613→   613→  const [project, clips, refs] = await Promise.all([
   614→   614→    db.project.findUnique({ where: { id: projectId } }),
   615→   615→    db.timelineClip.findMany({
   616→   616→      where: { projectId, ...(onlyClipIds ? { id: { in: onlyClipIds } } : {}) },
   617→   617→      orderBy: { order: "asc" },
   618→   618→    }),
   619→   619→    db.shotRef.findMany({
   620→   620→      where: { shot: { projectId } },
   621→   621→      include: { asset: true },
   622→   622→    }),
   623→   623→  ]);
   624→   624→  if (!project) throw new RouteError("项目不存在或已被删除", 404);
   625→   625→
   626→   626→  const refsByClip = new Map<string, typeof refs>();
   627→   627→  for (const r of refs) {
   628→   628→    const list = refsByClip.get(r.shotId) ?? [];
   629→   629→    list.push(r);
   630→   630→    refsByClip.set(r.shotId, list);
   631→   631→  }
   632→   632→
   633→   633→  let anchored = 0;
   634→   634→  let skipped = 0;
   635→   635→  for (const clip of clips) {
   636→   636→    if (clip.promptVersion === "manual") {
   637→   637→      skipped++;
   638→   638→      continue;
   639→   639→    }
   640→   640→    const bound = refsByClip.get(clip.id) ?? [];
   641→   641→    const characters = bound
   642→   642→      .filter((r) => r.asset.kind === "character")
   643→   643→      .map((r) => ({
   644→   644→        name: r.asset.name,
   645→   645→        visualAnchor: r.asset.visualAnchor,
   646→   646→        stateNote: r.stateNote,
   647→   647→      }));
   648→   648→    const sceneRef = bound.find((r) => r.asset.kind === "scene");
   649→   649→    const props = bound
   650→   650→      .filter((r) => r.asset.kind === "prop")
   651→   651→      .map((r) => ({
   652→   652→        name: r.asset.name,
   653→   653→        visualAnchor: r.asset.visualAnchor,
   654→   654→        stateNote: r.stateNote,
   655→   655→      }));
   656→   656→
   657→   657→    const { finalPrompt, fingerprint } = composeVideoPrompt({
   658→   658→      shot: {
   659→   659→        visualDesc: clip.visualDesc ?? clip.name,
   660→   660→        visualPrompt: clip.visualPrompt,
   661→   661→        cameraMovement: clip.cameraMovement,
   662→   662→        shotSize: clip.shotSize,
   663→   663→        subtitle: clip.subtitle,
   664→   664→      },
   665→   665→      characters,
   666→   666→      scene: sceneRef
   667→   667→        ? {
   668→   668→            name: sceneRef.asset.name,
   669→   669→            visualAnchor: sceneRef.asset.visualAnchor,
   670→   670→            stateNote: sceneRef.stateNote,
   671→   671→          }
   672→   672→        : null,
   673→   673→      props,
   674→   674→      tonePrompt: project.tonePrompt,
   675→   675→      aspectRatio: project.aspectRatio,
   676→   676→      plannedMs: clip.plannedMs,
   677→   677→      resolution: project.resolution,
   678→   678→    });
   679→   679→
   680→   680→    await db.timelineClip.update({
   681→   681→      where: { id: clip.id },
   682→   682→      data: { finalPrompt, promptVersion: fingerprint },
   683→   683→    });
   684→   684→    anchored++;
   685→   685→  }
   686→   686→  return { anchored, skipped };
   687→   687→}

   736→   736→async function executeRun(runId: string): Promise<void> {
   737→   737→  try {
   738→   738→    const run = await db.productionRun.findUnique({ where: { id: runId } });
   739→   739→    if (!run || run.status !== "running") return;
   740→   740→
   741→   741→    for (const def of RUN_STAGES) {
   742→   742→      const current = await loadRun(runId);
   743→   743→      if (current.status !== "running") return; // 被外部取消/覆盖
   744→   744→      const stages = parseStages(current.stages);
   745→   745→      const stage = stages.find((s) => s.key === def.key);
   746→   746→      if (!stage || stage.status === "succeeded") continue;
   747→   747→
   748→   748→      // —— 阶段置 running ——
   749→   749→      stage.status = "running";
   750→   750→      stage.startedAt = new Date().toISOString();
   751→   751→      stage.error = null;
   752→   752→      await patchStages(runId, stages);
   753→   753→      await db.productionRun.update({
   754→   754→        where: { id: runId },
   755→   755→        data: { stage: def.key },
   756→   756→      });
   757→   757→      const project = await db.project.findUnique({
   758→   758→        where: { id: current.projectId },
   759→   759→      });
   760→   760→      if (!project) throw new Error("项目不存在或已被删除");
   761→   761→
   762→   762→      const startMsg =
   763→   763→        STAGE_START_MESSAGE[def.key] ?? `进入「${def.name}」阶段`;
   764→   764→      const modeLabel =
   765→   765→        def.key === "screenwriting" ? thinkingModeLabel(project.thinkingLevel) : "";
   766→   766→      await appendLog(
   767→   767→        runId,
   768→   768→        def.role,
   769→   769→        startMsg.replace("{mode}", modeLabel)
   770→   770→      );
   771→   771→
   772→   772→      try {
   773→   773→        const result = await STAGE_FN[def.key](runId, project);
   774→   774→        const fresh = parseStages((await loadRun(runId)).stages);
   775→   775→        const stageRow = fresh.find((s) => s.key === def.key);
   776→   776→        if (stageRow) {
   777→   777→          stageRow.status = "succeeded";
   778→   778→          stageRow.finishedAt = new Date().toISOString();
   779→   779→          await patchStages(runId, fresh);
   780→   780→        }
   781→   781→        await appendLog(runId, def.role, result.summary);
   782→   782→        await recomputeProgress(runId);
   783→   783→      } catch (err) {
   784→   784→        const message =
   785→   785→          err instanceof Error ? err.message : "未知错误";
   786→   786→        const fresh = parseStages((await loadRun(runId)).stages);
   787→   787→        const stageRow = fresh.find((s) => s.key === def.key);
   788→   788→        if (stageRow) {
   789→   789→          stageRow.status = "failed";
   790→   790→          stageRow.finishedAt = new Date().toISOString();
   791→   791→          stageRow.error = message;
   792→   792→          await patchStages(runId, fresh);
   793→   793→        }
   794→   794→        await db.productionRun.update({
   795→   795→          where: { id: runId },
   796→   796→          data: { status: "failed", error: `「${def.name}」阶段失败：${message}` },
   797→   797→        });
   798→   798→        await appendLog(runId, def.role, `阶段失败：${message}`);
   799→   799→        return;
   800→   800→      }
   801→   801→    }
   802→   802→
   803→   803→    await db.productionRun.update({
   804→   804→      where: { id: runId },
   805→   805→      data: { status: "succeeded", progress: 100 },
   806→   806→    });

   850→   850→        },

   860→   860→export async function retryRun(runId: string): Promise<RunDTO> {
   861→   861→  const run = await loadRun(runId);
   862→   862→  if (run.status === "running") {
   863→   863→    throw new RouteError("流水线正在运行中，无需重试", 409);
   864→   864→  }
   865→   865→  const stages = parseStages(run.stages).map((s) =>
   866→   866→    s.status === "failed"
   867→   867→      ? { ...s, status: "pending", error: null, startedAt: null, finishedAt: null }
   868→   868→      : s
   869→   869→  );
   870→   870→  const resume = stages.find((s) => s.status !== "succeeded");
   871→   871→  await db.productionRun.update({
   872→   872→    where: { id: runId },
   873→   873→    data: {
   874→   874→      status: "running",
   875→   875→      error: null,
   876→   876→      stages: JSON.stringify(stages),
   877→   877→    },
   878→   878→  });
   879→   879→  await appendLog(
   880→   880→    runId,
   881→   881→    "producer",
   882→   882→    resume
   883→   883→      ? `断点续拍：从「${resume.name}」阶段继续（已完成阶段不重复执行）`
   884→   884→      : "复核完成，直接进入收尾"
   885→   885→  );
   886→   886→  void executeRun(runId);
   887→   887→  const fresh = await loadRun(runId);
   888→   888→  return toRunDTO(fresh);
   889→   889→}
