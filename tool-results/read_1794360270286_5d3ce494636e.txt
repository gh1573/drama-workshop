   158→/** 本地时间戳，格式 2026/11/10 13:58:19 */
   159→export function formatLocalTime(d: Date = new Date()): string {
   160→  const p = (n: number) => String(n).padStart(2, "0");
   161→  return `${d.getFullYear()}/${p(d.getMonth() + 1)}/${p(d.getDate())} ${p(
   162→    d.getHours()
   163→  )}:${p(d.getMinutes())}:${p(d.getSeconds())}`;
   164→}
   165→
   166→function pad2(n: number): string {
   167→  return String(n).padStart(2, "0");
   168→}

   170→/** 生成与原版完全对齐的 Markdown 制作清单 */
   171→export function buildMarkdown(project: DramaProjectData): string {
   172→  const p = project.project;
   173→  const lines: string[] = [];
   174→
   175→  lines.push("# 🎬 《剧集智能创编工坊》分镜制作与录音清单");
   176→  lines.push("");
   177→  lines.push(`> 导出时间：${formatLocalTime(new Date(p.exportedAt))}  `);
   178→  lines.push(`> 题材类型：**${p.genre.name}**（${p.genre.badge}）  `);
   179→  lines.push(`> 画幅比例：**${p.aspectRatio} 竖屏短剧**  `);
   180→  lines.push(
   181→    `> AI 视频基模：**${p.modelConfig.videoModel}** (${p.modelConfig.durationSeconds}s / ${p.modelConfig.resolution})  `
   182→  );
   183→  lines.push(`> 建议配乐音效：${p.genre.musicMood}`);
   184→  lines.push("");
   185→  lines.push("---");
   186→  lines.push("");
   187→  lines.push("## 📖 核心剧情梗概");
   188→  lines.push(p.synopsis);
   189→  lines.push("");
   190→  lines.push("");
   191→  lines.push("## ⚡ 创作破局技与钩子设定");
   192→  lines.push(`- **黄金前3秒抓人策略**：${p.genre.hookTactic}`);
   193→  lines.push(`- **画面影调 Prompt**：\`${p.genre.tonePrompt}\``);
   194→  lines.push("");
   195→  lines.push("");
   196→  lines.push("## 📋 三幕叙事与分镜详单");
   197→  lines.push("");
   198→
   199→  for (const beat of project.storyBeats) {
   200→    lines.push(`### 【第 ${pad2(beat.index)} 幕】${beat.actTitle}`);
   201→    lines.push(`- **节奏要求**：${beat.pacingNote}`);
   202→    lines.push(
   203→      `- **运镜机位**：${cameraLabel(beat.cameraMovement)}（\`${beat.cameraMovement}\` → ${CAMERA_MOVEMENTS[beat.cameraMovement]?.prompt ?? ""}）`
   204→    );
   205→    lines.push(`- **情节画面**：${beat.description}`);
   206→    lines.push(`- **核心台词金句 (字幕)**：> 💬 **${beat.dialogueSubtitle}**`);
   207→    lines.push(`- **画面提示补充**：${beat.promptGuidance}`);
   208→    lines.push(`- **镜头产出状态**：⏳ 待生成`);
   209→    lines.push("");
   210→    lines.push("");
   211→  }
   212→
   213→  // —— v2：资产定妆表（一致性锚定的交付物） ——
   214→  if (project.assets && project.assets.length > 0) {
   215→    lines.push("");
   216→    lines.push("---");
   217→    lines.push("");
   218→    lines.push(`## 🎭 资产定妆表 (${project.assets.length} 项 · 跨镜头一致性锚定)`);
   219→    lines.push("");
   220→    lines.push("| 类型 | 名称 | 锁定外观（中文） | 视觉锚点（英文） | 引用镜头数 | 定妆照 |");
   221→    lines.push("| :--- | :--- | :--- | :--- | :--- | :--- |");
   222→    const kindLabel: Record<string, string> = {
   223→      character: "角色",
   224→      scene: "场景",
   225→      prop: "道具",
   226→    };
   227→    for (const a of project.assets) {
   228→      lines.push(
   229→        `| ${kindLabel[a.kind] ?? a.kind} | **${a.name}** | ${a.lockedAppearance} | \`${a.visualAnchor}\` | ${a.refCount ?? 0} | ${a.portraitUrl ? "✅ 已定妆" : "⏳ 未定妆"} |`
   230→      );
   231→    }
   232→  }
   233→
   234→  lines.push("");
   235→  lines.push("---");
   236→  lines.push("");
   237→  lines.push(`## 🎞️ 镜头轨素材排布 (${project.timelineClips.length} 个片段)`);
   238→  lines.push("");
   239→  const hasBreakdown = project.timelineClips.some(
   240→    (c) => (c as ClipData).shotSize || (c as ClipData).cameraMovement
   241→  );
   242→  if (hasBreakdown) {
   243→    lines.push(
   244→      "| 镜头序号 | 镜头名称 | 关联节拍 | 景别 | 运镜机位 | 画面内容 | 台词字幕 | 计划时长 | 能量 | 音效 | 媒体格式 | 状态 |"
   245→    );
   246→    lines.push("| :--- | :--- | :--- | :--- | :--- | :--- | :--- | :--- | :--- | :--- | :--- | :--- |");
   247→    for (const clip of project.timelineClips) {
   248→      const c = clip as Record<string, string & number>;
   249→      const mv = c.cameraMovement
   250→        ? `${cameraLabel(String(c.cameraMovement))}`
   251→        : "—";
   252→      lines.push(
   253→        `| ${c.no ?? ""} | ${c.name ?? ""} | ${c.beatRef ?? ""} | ${c.shotSize ?? "—"} | ${mv} | ${c.visualDesc ?? "—"} | ${c.subtitle ?? ""} | ${c.plannedMs ? `${Math.round(c.plannedMs / 100) / 10}s` : "—"} | ${c.energy != null ? `${c.energy}` : "—"} | ${c.sfxHint ?? "—"} | ${c.format ?? ""} | ${c.status ?? ""} |`
   254→      );
   255→    }
   256→  } else {
   257→    lines.push(
   258→      "| 镜头序号 | 镜头名称 | 关联节拍 / 角色 | 台词字幕 | 媒体格式 |"
   259→    );
   260→    lines.push("| :--- | :--- | :--- | :--- | :--- |");
   261→    for (const clip of project.timelineClips) {
   262→      const c = clip as Record<string, string>;
   263→      lines.push(
   264→        `| ${c.no ?? ""} | ${c.name ?? ""} | ${c.beatRef ?? ""} | ${c.subtitle ?? ""} | ${c.format ?? ""} |`
   265→      );
   266→    }
   267→  }
   268→  lines.push("");
   269→  lines.push("");
   270→
   271→  // —— v2：三轨排布表（后期成包交付物） ——
   272→  if (project.tracks) {
   273→    const t = project.tracks;
   274→    const fmtMs = (ms: number) => `${Math.round(ms / 100) / 10}s`;
   275→    if (t.subtitle.length > 0 || t.sfx.length > 0 || t.bgm.length > 0) {
   276→      lines.push("---");
   277→      lines.push("");
   278→      lines.push("## 🎼 后期三轨排布");
   279→      lines.push("");
   280→      lines.push("### 字幕轨（金句大字 / 关键词变色）");
   281→      lines.push("| 入点 | 出点 | 字幕文本 | 样式 |");
   282→      lines.push("| :--- | :--- | :--- | :--- |");
   283→      for (const s of t.subtitle) {
   284→        lines.push(
   285→          `| ${fmtMs(s.startMs)} | ${fmtMs(s.endMs)} | ${s.text} | ${s.style ? JSON.stringify(s.style) : "—"} |`
   286→        );
   287→      }
   288→      lines.push("");
   289→      lines.push("### 音效轨（卡点音效）");
   290→      lines.push("| 入点 | 出点 | 音效标签 |");
   291→      lines.push("| :--- | :--- | :--- |");
   292→      for (const s of t.sfx) {
   293→        lines.push(`| ${fmtMs(s.startMs)} | ${fmtMs(s.endMs)} | ${s.text} |`);
   294→      }
   295→      lines.push("");
   296→      lines.push("### 配乐轨（情绪分段）");
   297→      lines.push("| 入点 | 出点 | 情绪标签 | 强度 |");
   298→      lines.push("| :--- | :--- | :--- | :--- |");
   299→      for (const s of t.bgm) {
   300→        const intensity =
   301→          s.style && typeof s.style === "object" && "intensity" in s.style
   302→            ? String((s.style as { intensity: number }).intensity)
   303→            : "—";
   304→        lines.push(`| ${fmtMs(s.startMs)} | ${fmtMs(s.endMs)} | ${s.text} | ${intensity} |`);
   305→      }
   306→    }
   307→  }
   308→
   309→  // —— v2：镜头级最终提示词清单（可直接投喂 AI 视频基模） ——
   310→  const prompts = project.timelineClips.filter(
   311→    (c) => (c as Record<string, unknown>).finalPrompt
   312→  );
   313→  if (prompts.length > 0) {
   314→    lines.push("");
   315→    lines.push("---");
   316→    lines.push("");
   317→    lines.push(`## 🤖 镜头级最终提示词 (${prompts.length} 条 · 锚定组装产物)`);
   318→    lines.push("");
   319→    for (const clip of prompts) {
   320→      const c = clip as Record<string, string>;
   321→      lines.push(
   322→        `**#${c.no} ${c.name}**（${c.shotSize ?? ""} · ${cameraLabel(c.cameraMovement ?? "")}）`
   323→      );
   324→      lines.push("```text");
   325→      lines.push(c.finalPrompt);
   326→      lines.push("```");
   327→      lines.push("");
   328→    }
   329→  }
   330→
   331→  lines.push("");
   332→  lines.push("---");
   333→  lines.push("*制作团队：剧集智能创编工坊 (DRAMA STUDIO) · AI 剧组流水线*");
   334→
   335→  return lines.join("\n");
   336→}

   338→/** 触发浏览器下载 */
   339→export function downloadFile(filename: string, content: string, mime: string) {
   340→  const blob = new Blob([content], { type: mime });
   341→  const url = URL.createObjectURL(blob);
   342→  const a = document.createElement("a");
   343→  a.href = url;
   344→  a.download = filename;
   345→  document.body.appendChild(a);
   346→  a.click();
   347→  document.body.removeChild(a);
   348→  URL.revokeObjectURL(url);
   349→}

   825→/**
   826→ * 将项目详情转换为与原版对齐的导出工程数据。
   827→ * timelineClips 按 order 排序填入，含状态与成片地址（原版 schema 的前向兼容超集）。
   828→ */
   829→export function toExportData(detail: ProjectDetail): DramaProjectData {
   830→  const clips = [...detail.timelineClips].sort((a, b) => a.order - b.order);
   831→  return {
   832→    project: {
   833→      title: detail.title,
   834→      version: detail.schemaVersion,
   835→      exportedAt: new Date().toISOString(),
   836→      genre: { ...detail.genre },
   837→      aspectRatio: detail.aspectRatio,
   838→      synopsis: detail.synopsis,
   839→      modelConfig: detail.modelConfig,
   840→    },
   841→    storyBeats: detail.storyBeats,
   842→    timelineClips: clips.map((c, i) => ({
   843→      no: i + 1,
   844→      name: c.name,
   845→      beatRef: `第 ${String(c.beatIndex).padStart(2, "0")} 幕`,
   846→      subtitle: c.subtitle,
   847→      format: c.format,
   848→      status: CLIP_STATUS[c.status].label,
   849→      mediaUrl: c.mediaUrl ?? "",
   850→      ...(c.shotSize ? { shotSize: c.shotSize } : {}),
   851→      ...(c.cameraMovement
   852→        ? {
   853→            cameraMovement: c.cameraMovement,
   854→            cameraPrompt: CAMERA_MOVEMENTS[c.cameraMovement]?.prompt ?? "",
   855→          }
   856→        : {}),
   857→      ...(c.visualDesc ? { visualDesc: c.visualDesc } : {}),
   858→      ...(c.plannedMs != null ? { plannedMs: c.plannedMs } : {}),
   859→      ...(c.finalPrompt ? { finalPrompt: c.finalPrompt } : {}),
   860→      ...(c.energy != null ? { energy: c.energy } : {}),
   861→      ...(c.sfxHint ? { sfxHint: c.sfxHint } : {}),
   862→    })),
   863→    statistics: {
   864→      totalBeats: detail.storyBeats.length,
   865→      totalClips: clips.length,
   866→      hasSynthesizedOutput: clips.length > 0 && clips.every((c) => c.status === "ready"),
   867→    },
   868→    // —— v2 可选扩展段（旧消费者忽略未知字段即可） ——
   869→    ...(detail.assets.length > 0
   870→      ? {
   871→          assets: detail.assets.map((a) => ({
   872→            kind: a.kind,
   873→            name: a.name,
   874→            lockedAppearance: a.lockedAppearance,
   875→            visualAnchor: a.visualAnchor,
   876→            ...(a.portraitUrl ? { portraitUrl: a.portraitUrl } : {}),
   877→            refCount: a.refCount,
   878→          })),
   879→        }
   880→      : {}),
   881→    ...(detail.tracks.length > 0
   882→      ? {
   883→          tracks: {
   884→            subtitle: detail.tracks.filter((t) => t.kind === "subtitle"),
   885→            sfx: detail.tracks.filter((t) => t.kind === "sfx"),
   886→            bgm: detail.tracks.filter((t) => t.kind === "bgm"),
   887→          },
   888→        }
   889→      : {}),
   890→    ...(clips.some((c) => c.energy != null)
   891→      ? {
   892→          energyCurve: clips.map((c, i) => ({
   893→            clipId: c.id,
   894→            order: i,
   895→            beatIndex: c.beatIndex,
   896→            energy: c.energy ?? 50,
   897→          })),
   898→        }
   899→      : {}),
   900→  };
   901→}
