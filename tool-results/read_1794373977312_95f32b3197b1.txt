    60→function parseStages(raw: string): RunStageInfo[] {
    61→  try {
    62→    const parsed = JSON.parse(raw) as RunStageInfo[];
    63→    if (Array.isArray(parsed) && parsed.length === RUN_STAGES.length) {
    64→      return parsed;
    65→    }
    66→  } catch {
    67→    /* fallthrough */
    68→  }
    69→  return freshStages();
    70→}

    73→  try {

    75→    if (Array.isArray(parsed)) return parsed;
    76→  } catch {
    77→    /* fallthrough */
    78→  }
    79→  return [];
    80→}

   148→async function recomputeProgress(runId: string): Promise<void> {
   149→  const row = await loadRun(runId);
   150→  const stages = parseStages(row.stages);
   151→  const done = stages
   152→    .filter((s) => s.status === "succeeded")
   153→    .reduce(
   154→      (sum, s) =>
   155→        sum + (RUN_STAGES.find((d) => d.key === s.key)?.weight ?? 0),
   156→      0
   157→    );
   158→  const total = RUN_STAGES.reduce((sum, d) => sum + d.weight, 0);
   159→  await db.productionRun.update({
   160→    where: { id: runId },
   161→    data: { progress: Math.min(100, Math.round((done / total) * 100)) },
   162→  });
   163→}

   256→function parseBeats(raw: string): StoryBeat[] | null {
   257→  const data = parseJsonLoose(raw) as { storyBeats?: unknown[] } | null;
   258→  const arr = data?.storyBeats;
   259→  if (!Array.isArray(arr) || arr.length < 3) return null;
   260→
   261→  const funcs: NarrativeFunction[] = ["hook", "climax", "cliffhanger"];
   262→  const beats: StoryBeat[] = [];
   263→  for (let i = 0; i < 3; i++) {
   264→    const item = arr[i] as Record<string, unknown>;
   265→    const fn = funcs.includes(item?.narrativeFunction as NarrativeFunction)
   266→      ? (item.narrativeFunction as NarrativeFunction)
   267→      : funcs[i];
   268→    const meta = NARRATIVE_META[fn];
   269→    const str = (v: unknown, fallback: string) =>
   270→      typeof v === "string" && v.trim() ? v.trim() : fallback;
   271→    const cam =
   272→      typeof item?.cameraMovement === "string" &&
   273→      item.cameraMovement in CAMERA_MOVEMENTS
   274→        ? item.cameraMovement
   275→        : "";
   276→    beats.push({
   277→      index: i + 1,
   278→      actTitle: `${meta.actNo}：${meta.label} (${meta.tag})`,
   279→      narrativeFunction: fn,
   280→      pacingNote: str(item?.pacingNote, PACING_FALLBACK[fn]),
   281→      description: str(item?.description, "（分镜内容生成异常，请重试）"),
   282→      dialogueSubtitle: str(item?.dialogueSubtitle, "台词生成异常，请重试"),
   283→      cameraMovement: cam,
   284→      promptGuidance: str(item?.promptGuidance, "提示生成异常，请重试"),
   285→      energyStart: null,
   286→      energyEnd: null,
   287→    });
   288→  }
   289→  // 运镜合规修复：池外 → 本幕池；三幕互斥 → 换池内备选
   290→  const used = new Set<string>();
   291→  for (const beat of beats) {
   292→    const pool = ACT_MOVEMENT_POOLS[beat.narrativeFunction];
   293→    let mv = beat.cameraMovement;
   294→    if (!pool.includes(mv)) mv = pool[0];
   295→    if (used.has(mv)) {
   296→      const alt = pool.find((k) => !used.has(k));
   297→      if (alt) mv = alt;
   298→    }
   299→    used.add(mv);
   300→    beat.cameraMovement = mv;
   301→  }
   302→  return beats;
   303→}

   305→async function stageScreenwriting(
   306→  runId: string,
   307→  project: Project
   308→): Promise<StageResult> {
   309→  const provider = getTextProvider();
   310→  const base = buildScreenwritingMessages(project);
   311→
   312→  let beats: StoryBeat[] | null = null;
   313→  let lastRaw = "";
   314→  for (let attempt = 0; attempt < 2 && !beats; attempt++) {
   315→    const { text } = await provider.complete({
   316→      messages: [
   317→        base.messages[0],
   318→        {
   319→          role: "user",
   320→          content:
   321→            attempt === 0
   322→              ? base.messages[1].content
   323→              : `${base.messages[1].content}\n\n注意：上一次输出无法解析。请严格只输出一个合法 JSON 对象（含 storyBeats 数组），不要任何其他文字。`,
   324→        },
   325→      ],
   326→      thinking: { type: project.thinkingLevel === "low" ? "disabled" : "enabled" },
   327→    });
   328→    lastRaw = text;
   329→    beats = parseBeats(text);
   330→  }
   331→  if (!beats) {
   332→    throw new Error(`剧本解析失败（${lastRaw.slice(0, 120)}…）`);
   333→  }
   334→  await updateProject(project.id, { storyBeats: beats, status: "beatsReady" });
   335→  return {
   336→    summary: `三幕剧本定稿：${beats
   337→      .map((b) => NARRATIVE_META[b.narrativeFunction].label)
   338→      .join(" → ")}，运镜 ${beats.map((b) => b.cameraMovement).join(" / ")}`,
   339→  };
   340→}

   352→function buildCastingMessages(beats: DbBeat[], project: Project): string {
   353→  const script = beats
   354→    .map(
   355→      (b) =>
   356→        `【第 ${b.index} 幕 · ${b.actTitle}】\n画面：${b.description}\n金句：${b.dialogueSubtitle}\n提示：${b.promptGuidance}`
   357→    )

   359→
   360→  return `你是微短剧剧组的选角导演兼美术指导。任务：通读三幕剧本，建立全剧资产库（角色 / 场景 / 道具），供后续所有镜头做「一致性锚定」——每个镜头的提示词都将引用这里的定妆描述，保证跨镜头不换脸、不穿帮。
   361→
   362→【题材调性】${project.genreName}（${project.genreBadge}）；英文影调锚点：${project.tonePrompt}
   363→
   364→${CASTING_FACE_RULE}
   365→
   366→【三幕剧本】
   367→${script}
   368→
   369→【要求（严格遵守）】
   370→1. characters：主角、对手与关键配角，最多 6 个。lockedAppearance 为中文定妆描述（25~60 字：年龄、性别、身份气质、发型、脸型特征、服装、标志物）；visualAnchor 为对应英文串（15~30 words，如 "35-year-old Chinese male special forces captain, buzz cut, scar on left brow, black tactical vest"）
   371→2. scenes：关键场景，最多 5 个。name 为短名（如「宴会厅」「冷宫」）；lockedAppearance 描述空间与光线；visualAnchor 为英文
   372→3. props：推动剧情的关键道具，最多 4 个，没有则输出空数组
   373→4. 名称必须与剧本用词一致，不得杜撰剧本中未出现的角色
   374→5. visualAnchor 是所有镜头共用的视觉锚点，聚焦「跨镜头不变的外观特征」，不要写动作
   375→
   376→【输出格式】只输出 JSON，禁止解释文字、禁止 markdown 围栏：
   377→{"characters":[{"name":"…","lockedAppearance":"…","visualAnchor":"…"}],"scenes":[{"name":"…","lockedAppearance":"…","visualAnchor":"…"}],"props":[{"name":"…","lockedAppearance":"…","visualAnchor":"…"}]}`;
   378→}
   379→
   380→function parseCasting(
   381→  raw: string
   382→): { characters: CastingAssetDraft[]; scenes: CastingAssetDraft[]; props: CastingAssetDraft[] } | null {
   383→  const data = parseJsonLoose(raw) as Record<string, unknown> | null;
   384→  if (!data) return null;
   385→
   386→  const map = (v: unknown, max: number): CastingAssetDraft[] =>
   387→    (Array.isArray(v) ? v : [])
   388→      .map((item) => {
   389→        const o = item as Record<string, unknown>;
   390→        const name = typeof o?.name === "string" ? o.name.trim().slice(0, 20) : "";
   391→        if (!name) return null;
   392→        const lockedAppearance =
   393→          typeof o?.lockedAppearance === "string" ? o.lockedAppearance.trim().slice(0, 200) : "";
   394→        const visualAnchor =
   395→          typeof o?.visualAnchor === "string" ? o.visualAnchor.trim().slice(0, 300) : "";
   396→        if (!lockedAppearance && !visualAnchor) return null;
   397→        return {
   398→          name,
   399→          lockedAppearance: lockedAppearance || "（定妆描述待补充）",
   400→          visualAnchor: visualAnchor || lockedAppearance,
   401→        } satisfies CastingAssetDraft;
   402→      })
   403→      .filter((x): x is CastingAssetDraft => !!x)
   404→      .slice(0, max);
   405→
   406→  const characters = map(data.characters, 6);
   407→  const scenes = map(data.scenes, 5);
   408→  const props = map(data.props, 4);
   409→  if (characters.length === 0 && scenes.length === 0) return null;
   410→  return { characters, scenes, props };
   411→}

   427→  let casting: ReturnType<typeof parseCasting> = null;
   428→  let lastRaw = "";
   429→  for (let attempt = 0; attempt < 2 && !casting; attempt++) {
   430→    const { text } = await provider.complete({
   431→      messages: [
   432→        { role: "assistant", content: prompt },
   433→        {
   434→          role: "user",
   435→          content:
   436→            attempt === 0
   437→              ? "请建立资产库并只输出 JSON。"
   438→              : "上一次输出无法解析。请严格只输出一个合法 JSON 对象（characters/scenes/props 三个数组），不要任何其他文字。",
   439→        },
   440→      ],
   441→      thinking: { type: "disabled" },
   442→    });
   443→    lastRaw = text;
   444→    casting = parseCasting(text);
   445→  }
   446→  if (!casting) {
   447→    throw new Error(`选角定妆解析失败（${lastRaw.slice(0, 120)}…）`);
   448→  }
   449→
   450→  // 幂等 upsert：同名资产更新锚点（保证与最新剧本一致）
   451→  const kindGroups: Array<[AssetKind, CastingAssetDraft[]]> = [
   452→    ["character", casting.characters],
   453→    ["scene", casting.scenes],
   454→    ["prop", casting.props],
   455→  ];
   456→  let count = 0;
   457→  for (const [kind, drafts] of kindGroups) {
   458→    for (const draft of drafts) {
   459→      await db.asset.upsert({
   460→        where: {
   461→          projectId_kind_name: {
   462→            projectId: project.id,
   463→            kind,
   464→            name: draft.name,
   465→          },
   466→        },
   467→        create: {
   468→          projectId: project.id,
   469→          kind,
   470→          name: draft.name,
   471→          lockedAppearance: draft.lockedAppearance,
   472→          visualAnchor: draft.visualAnchor,
   473→          source: "crew",
   474→        },
   475→        update: {
   476→          lockedAppearance: draft.lockedAppearance,
   477→          visualAnchor: draft.visualAnchor,
   478→        },
   479→      });
   480→      count++;
   481→    }
   482→  }
   483→  await appendLog(
   484→    runId,
   485→    "casting",
   486→    `资产入库：角色 ${casting.characters.length} · 场景 ${casting.scenes.length} · 道具 ${casting.props.length}`
   487→  );
   488→  return {
   489→    summary: `资产库建立：${count} 项定妆锁定（角色 ${casting.characters.length} / 场景 ${casting.scenes.length} / 道具 ${casting.props.length}）`,
   490→  };
   491→}

   497→async function stageDirecting(
   498→  runId: string,
   499→  project: Project
   500→): Promise<StageResult> {
   501→  const [beats, assets] = await Promise.all([
   502→    db.storyBeat.findMany({
   503→      where: { projectId: project.id },
   504→      orderBy: { index: "asc" },
   505→    }),
   506→    db.asset.findMany({ where: { projectId: project.id } }),
   507→  ]);
   508→  if (beats.length < 3) {
   509→    throw new Error("剧本尚未就绪，请先完成编剧阶段");
   510→  }
   511→
   512→  const roster = {
   513→    characters: assets.filter((a) => a.kind === "character").map((a) => a.name),
   514→    scenes: assets.filter((a) => a.kind === "scene").map((a) => a.name),
   515→    props: assets.filter((a) => a.kind === "prop").map((a) => a.name),
   516→  };
   517→
   518→  let totalShots = 0;
   519→  let skippedActs = 0;
   520→  for (const beat of beats) {
   521→    // 成片保护：该幕已有已生成镜头时跳过拆解（不覆盖劳动成果），
   522→    // 由日志明确告知；如需重拍请先在镜头轨把该幕镜头重置为待生成
   523→    const hasReadyClip = await db.timelineClip.findFirst({
   524→      where: { projectId: project.id, beatIndex: beat.index, status: "ready" },
   525→      select: { id: true },
   526→    });
   527→    if (hasReadyClip) {
   528→      skippedActs++;
   529→      await appendLog(
   530→        runId,
   531→        "director",
   532→        `第 ${beat.index} 幕已有已生成镜头，跳过拆解以保护成片（重拍请先重置该幕镜头）`
   533→      );
   534→      continue;
   535→    }
   536→    const shots = await breakdownAct({
   537→      genreName: project.genreName,
   538→      genreBadge: project.genreBadge,
   539→      tonePrompt: project.tonePrompt,
   540→      synopsis: project.synopsis,
   541→      beats: beats.map((b) => ({
   542→        index: b.index,
   543→        actTitle: b.actTitle,
   544→        narrativeFunction: b.narrativeFunction as StoryBeat["narrativeFunction"],
   545→        pacingNote: b.pacingNote,
   546→        description: b.description,
   547→        dialogueSubtitle: b.dialogueSubtitle,
   548→        cameraMovement: b.cameraMovement,
   549→        promptGuidance: b.promptGuidance,
   550→      })),
   551→      targetIndex: beat.index,
   552→      assetRoster: roster,
   553→    });
   554→    const clips = await replaceActClips(project.id, beat.index, shots);
   555→    await bindRefsFromShots(project.id, beat.index, shots, clips);
   556→    await updateBeatEnergy(project.id, beat.index, {
   557→      energyStart: shots[0]?.energy ?? null,
   558→      energyEnd: shots[shots.length - 1]?.energy ?? null,
   559→    });
   560→    totalShots += shots.length;
   561→    await appendLog(
   562→      runId,
   563→      "director",
   564→      `第 ${beat.index} 幕拆解完成：${shots.length} 镜（${shots
   565→        .map((s) => s.shotSize)
   566→        .join("→")}，能量 ${shots[0]?.energy ?? "-"}→${shots[shots.length - 1]?.energy ?? "-"}）`
   567→    );
   568→  }
   569→  return {
   570→    summary: `三幕拆镜完成：共 ${totalShots} 个镜头，正反打景别交替已就位${
   571→      skippedActs > 0 ? `；${skippedActs} 幕含成片已跳过` : ""
   572→    }`,
   573→  };
   574→}

   576→/** 按名称把拆镜产物中的资产引用解析为 ShotRef（精确名 → 包含式回退） */
   577→export async function bindRefsFromShots(
   578→  projectId: string,
   579→  beatIndex: number,
   580→  shots: ShotBreakdownItem[],
   581→  clips: Array<{ id: string; beatIndex: number; order: number }>
   582→): Promise<number> {
   583→  const assets = await db.asset.findMany({ where: { projectId } });
   584→  const byKind = (kind: AssetKind) => assets.filter((a) => a.kind === kind);
   585→  const resolve = (kind: AssetKind, name: string) => {
   586→    if (!name) return null;
   587→    const pool = byKind(kind);
   588→    return (
   589→      pool.find((a) => a.name === name) ??
   590→      pool.find(
   591→        (a) => a.name.includes(name) || name.includes(a.name)
   592→      ) ??
   593→      null
   594→    );
   595→  };
   596→
   597→  const actClips = clips
   598→    .filter((c) => c.beatIndex === beatIndex)
   599→    .sort((a, b) => a.order - b.order);
   600→  let bound = 0;
   601→  for (let i = 0; i < actClips.length && i < shots.length; i++) {
   602→    const shot = shots[i];
   603→    const clip = actClips[i];
   604→    const targets = new Set<string>();
   605→    for (const name of shot.characters) {
   606→      const asset = resolve("character", name);
   607→      if (asset) targets.add(asset.id);
   608→    }
   609→    const scene = resolve("scene", shot.scene);
   610→    if (scene) targets.add(scene.id);
   611→    for (const name of shot.props) {
   612→      const asset = resolve("prop", name);
   613→      if (asset) targets.add(asset.id);
   614→    }
   615→    for (const assetId of targets) {
   616→      await db.shotRef.upsert({
   617→        where: { shotId_assetId: { shotId: clip.id, assetId } },
   618→        create: { shotId: clip.id, assetId },
   619→        update: {},
   620→      });
   621→      bound++;
   622→    }
   623→  }
   624→  return bound;
   625→}

   631→export async function anchorProjectClips(
   632→  projectId: string,
   633→  onlyClipIds?: string[]
   634→): Promise<{ anchored: number; skipped: number }> {
   635→  const [project, clips, refs] = await Promise.all([
   636→    db.project.findUnique({ where: { id: projectId } }),
   637→    db.timelineClip.findMany({
   638→      where: { projectId, ...(onlyClipIds ? { id: { in: onlyClipIds } } : {}) },
   639→      orderBy: { order: "asc" },
   640→    }),
   641→    db.shotRef.findMany({
   642→      where: { shot: { projectId } },
   643→      include: { asset: true },
   644→    }),
   645→  ]);
   646→  if (!project) throw new RouteError("项目不存在或已被删除", 404);
   647→
   648→  const refsByClip = new Map<string, typeof refs>();
   649→  for (const r of refs) {
   650→    const list = refsByClip.get(r.shotId) ?? [];
   651→    list.push(r);
   652→    refsByClip.set(r.shotId, list);
   653→  }
   654→
   655→  let anchored = 0;
   656→  let skipped = 0;
   657→  for (const clip of clips) {
   658→    if (clip.promptVersion === "manual") {
   659→      skipped++;
   660→      continue;
   661→    }
   662→    const bound = refsByClip.get(clip.id) ?? [];
   663→    const characters = bound
   664→      .filter((r) => r.asset.kind === "character")
   665→      .map((r) => ({
   666→        name: r.asset.name,
   667→        visualAnchor: r.asset.visualAnchor,
   668→        stateNote: r.stateNote,
   669→      }));
   670→    const sceneRef = bound.find((r) => r.asset.kind === "scene");
   671→    const props = bound
   672→      .filter((r) => r.asset.kind === "prop")
   673→      .map((r) => ({
   674→        name: r.asset.name,
   675→        visualAnchor: r.asset.visualAnchor,
   676→        stateNote: r.stateNote,
   677→      }));
   678→
   679→    const { finalPrompt, fingerprint } = composeVideoPrompt({
   680→      shot: {
   681→        visualDesc: clip.visualDesc ?? clip.name,
   682→        visualPrompt: clip.visualPrompt,
   683→        cameraMovement: clip.cameraMovement,
   684→        shotSize: clip.shotSize,
   685→        subtitle: clip.subtitle,
   686→      },
   687→      characters,
   688→      scene: sceneRef
   689→        ? {
   690→            name: sceneRef.asset.name,
   691→            visualAnchor: sceneRef.asset.visualAnchor,
   692→            stateNote: sceneRef.stateNote,
   693→          }
   694→        : null,
   695→      props,
   696→      tonePrompt: project.tonePrompt,
   697→      aspectRatio: project.aspectRatio,
   698→      plannedMs: clip.plannedMs,
   699→      resolution: project.resolution,
   700→    });
   701→
   702→    await db.timelineClip.update({
   703→      where: { id: clip.id },
   704→      data: { finalPrompt, promptVersion: fingerprint },
   705→    });
   706→    anchored++;
   707→  }
   708→  return { anchored, skipped };
   709→}

   758→async function executeRun(runId: string): Promise<void> {
   759→  try {
   760→    const run = await db.productionRun.findUnique({ where: { id: runId } });
   761→    if (!run || run.status !== "running") return;
   762→
   763→    for (const def of RUN_STAGES) {
   764→      const current = await loadRun(runId);
   765→      if (current.status !== "running") return; // 被外部取消/覆盖
   766→      const stages = parseStages(current.stages);
   767→      const stage = stages.find((s) => s.key === def.key);
   768→      if (!stage || stage.status === "succeeded") continue;
   769→
   770→      // —— 阶段置 running ——
   771→      stage.status = "running";
   772→      stage.startedAt = new Date().toISOString();
   773→      stage.error = null;
   774→      await patchStages(runId, stages);
   775→      await db.productionRun.update({
   776→        where: { id: runId },
   777→        data: { stage: def.key },
   778→      });
   779→      const project = await db.project.findUnique({
   780→        where: { id: current.projectId },
   781→      });
   782→      if (!project) throw new Error("项目不存在或已被删除");
   783→
   784→      const startMsg =
   785→        STAGE_START_MESSAGE[def.key] ?? `进入「${def.name}」阶段`;
   786→      const modeLabel =
   787→        def.key === "screenwriting" ? thinkingModeLabel(project.thinkingLevel) : "";
   788→      await appendLog(
   789→        runId,
   790→        def.role,
   791→        startMsg.replace("{mode}", modeLabel)
   792→      );
   793→
   794→      try {
   795→        const result = await STAGE_FN[def.key](runId, project);
   796→        const fresh = parseStages((await loadRun(runId)).stages);
   797→        const stageRow = fresh.find((s) => s.key === def.key);
   798→        if (stageRow) {
   799→          stageRow.status = "succeeded";
   800→          stageRow.finishedAt = new Date().toISOString();
   801→          await patchStages(runId, fresh);
   802→        }
   803→        await appendLog(runId, def.role, result.summary);
   804→        await recomputeProgress(runId);
   805→      } catch (err) {
   806→        const message =
   807→          err instanceof Error ? err.message : "未知错误";
   808→        const fresh = parseStages((await loadRun(runId)).stages);
   809→        const stageRow = fresh.find((s) => s.key === def.key);
   810→        if (stageRow) {
   811→          stageRow.status = "failed";
   812→          stageRow.finishedAt = new Date().toISOString();
   813→          stageRow.error = message;
   814→          await patchStages(runId, fresh);
   815→        }
   816→        await db.productionRun.update({
   817→          where: { id: runId },
   818→          data: { status: "failed", error: `「${def.name}」阶段失败：${message}` },
   819→        });
   820→        await appendLog(runId, def.role, `阶段失败：${message}`);
   821→        return;
   822→      }
   823→    }
   824→
   825→    await db.productionRun.update({
   826→      where: { id: runId },
   827→      data: { status: "succeeded", progress: 100 },
   828→    });
   829→    await appendLog(
   830→      runId,
   831→      "producer",
   832→      "全片杀青 🎬 制作包已就绪：三幕剧本 · 分镜表 · 资产定妆 · 镜头级提示词 · 三轨 · 能量曲线，可导出交付"
   833→    );
   834→  } catch (err) {
   835→    const message = err instanceof Error ? err.message : "未知错误";
   836→    await db.productionRun
   837→      .update({
   838→        where: { id: runId },
   839→        data: { status: "failed", error: message },
   840→      })
   841→      .catch(() => undefined);
   842→  }
   843→}

   881→/** 断点续跑：已成功阶段保留，从首个未成功阶段继续 */
   882→export async function retryRun(runId: string): Promise<RunDTO> {
   883→  const run = await loadRun(runId);
   884→  if (run.status === "running") {
   885→    throw new RouteError("流水线正在运行中，无需重试", 409);
   886→  }
   887→  const stages = parseStages(run.stages).map((s) =>
   888→    s.status === "failed"
   889→      ? { ...s, status: "pending", error: null, startedAt: null, finishedAt: null }
   890→      : s
   891→  );
   892→  const resume = stages.find((s) => s.status !== "succeeded");
   893→  await db.productionRun.update({
   894→    where: { id: runId },
   895→    data: {
   896→      status: "running",
   897→      error: null,
   898→      stages: JSON.stringify(stages),
   899→    },
   900→  });
   901→  await appendLog(
   902→    runId,
   903→    "producer",
   904→    resume
   905→      ? `断点续拍：从「${resume.name}」阶段继续（已完成阶段不重复执行）`
   906→      : "复核完成，直接进入收尾"
   907→  );
   908→  void executeRun(runId);
   909→  const fresh = await loadRun(runId);
   910→  return toRunDTO(fresh);
   911→}
